%% Dạng 1. Thực hiện phép tính — 4 ý: a cơ bản, b tính hợp lý, c và d nhiều ngoặc và lũy thừa.
\begin{bt} Thực hiện phép tính:
	\begin{listEX}[2]
		\item $(-27)+15-(-6)$;
		\item $(-24)\cdot 15+(-24)\cdot 35$;
		\item $\left[(-36):4+2\cdot(-5)\right]:(-1)^{2}+(-2)^{3}$;
		\item $540-18:\left[(8-5)^{2}+(-2)^{3}\cdot 5^{0}\right]$.
	\end{listEX}
	\loigiai{
		\begin{listEX}
			\item $(-27)+15-(-6)=(-27)+15+6= -6$.
			\item $(-24)\cdot 15+(-24)\cdot 35=(-24)\cdot(15+35)=(-24)\cdot 50=-1200$.
			\item $\left[(-36):4+2\cdot(-5)\right]:(-1)^{2}+(-2)^{3}$\\
			$=\left[-9+(-10)\right]:1+(-8)$\\
			$=(-19)+(-8)=-27$.
			\item $540-18:\left[(8-5)^{2}+(-2)^{3}\cdot 5^{0}\right]$\\
			$=540-18:\left[9+(-8)\cdot 1\right]$\\
			$=540-18:(9-8)$\\
			$=540-18:1=522$.
		\end{listEX}
	}
\end{bt}

%% Dạng 2. Tìm x — 4 ý: a cơ bản, b trị tuyệt đối, c ngoặc lồng, d có x ở cả hai vế.
\begin{bt} Tìm số nguyên $x$, biết:
	\begin{listEX}
		\item $8x=-56$;
		\item $|3x+6|=15$;
		\item $4\left[2(x-3)+5\right]-6=30$;
		\item $4(x+3)-3(x-1)=2x+5$.
	\end{listEX}
	\loigiai{
		\begin{listEX}
			\item $8x=-56$, suy ra $x=-56:8=-7$.
			\item $|3x+6|=15$ nên có hai trường hợp.\\
			Trường hợp 1: $3x+6=15$, nên $3x=9$, $x=3$.\\
			Trường hợp 2: $3x+6=-15$, nên $3x=-21$, $x=-7$.\\
			Vậy $x=3$ hoặc $x=-7$.
			\item $4\left[2(x-3)+5\right]-6=30$\\
			$4(2x-6+5)-6=30$\\
			$4(2x-1)-6=30$\\
			$8x-4-6=30$\\
			$8x-10=30$\\
			$8x=40$\\
			$x=5$.
			\item $4(x+3)-3(x-1)=2x+5$\\
			$4x+12-3x+3=2x+5$\\
			$x+15=2x+5$\\
			$x=10$.
		\end{listEX}
	}
\end{bt}

%% Dạng 3. Tập hợp — 2 ý, số nguyên và số tự nhiên, nhiều điều kiện.
\begin{bt} Biết số $0$ là số tự nhiên. Gọi $M$ là tập hợp các số nguyên $x$ thỏa mãn đồng thời $-12\le x\le 12$, $x$ chia hết cho $3$ và $x$ không chia hết cho $2$. Gọi $N$ là tập hợp các số tự nhiên nhỏ hơn $15$ và chia hết cho $3$.
	\begin{enumerate}
		\item Liệt kê các phần tử của $M$.
		\item Liệt kê các số vừa thuộc $M$ vừa thuộc $N$. Liệt kê các số thuộc $N$ nhưng không thuộc $M$.
	\end{enumerate}
	\loigiai{
		\begin{enumerate}
			\item Các số nguyên từ $-12$ đến $12$ chia hết cho $3$ là\\
			$-12;\,-9;\,-6;\,-3;\,0;\,3;\,6;\,9;\,12$.\\
			Trong các số này, số không chia hết cho $2$ là $-9;\,-3;\,3;\,9$.\\
			Vậy $M=\{-9;\,-3;\,3;\,9\}$.
			\item Số vừa thuộc $M$ vừa thuộc $N$ là $3$ và $9$.\\
			$N=\{0;\,3;\,6;\,9;\,12\}$.\\
			Các số thuộc $N$ nhưng không thuộc $M$ là $0;\,6;\,12$.
		\end{enumerate}
	}
\end{bt}

%% Dạng 4. Ước và bội — thực tế, học sinh tự nhận ra ƯCLN.
\begin{bt} Một câu lạc bộ có $36$ bạn đăng ký múa và $48$ bạn đăng ký hát đồng ca. Thầy muốn chia thành các tổ luyện tập sao cho các tổ có số bạn múa bằng nhau, số bạn hát bằng nhau, mỗi tổ có ít nhất một bạn múa và một bạn hát, và mọi bạn đều được chia vào tổ.
	\begin{enumerate}
		\item Có bao nhiêu cách chia nếu số tổ lớn hơn $1$?
		\item Cách chia nào làm mỗi tổ có ít bạn nhất? Khi đó mỗi tổ có bao nhiêu bạn múa, bao nhiêu bạn hát?
	\end{enumerate}
	\loigiai{
		\begin{enumerate}
			\item Số tổ phải là ước chung của $36$ và $48$.\\
			$36=2^{2}\cdot 3^{2}$, $48=2^{4}\cdot 3$, nên ước chung lớn nhất là $2^{2}\cdot 3=12$.\\
			Các ước dương của $12$ là $1;\,2;\,3;\,4;\,6;\,12$.\\
			Vì số tổ lớn hơn $1$, có $5$ cách: chia thành $2$, $3$, $4$, $6$ hoặc $12$ tổ.
			\item Mỗi tổ ít bạn nhất khi số tổ nhiều nhất, tức là $12$ tổ.\\
			Mỗi tổ có $36:12=3$ bạn múa và $48:12=4$ bạn hát.
		\end{enumerate}
	}
\end{bt}

%% Dạng 5. Toán thực tế — BCNN, khoảng giá trị cho đúng một đáp án.
\begin{bt} Thư viện đóng số sách quyên góp được thành từng bó $12$ cuốn hoặc từng bó $18$ cuốn đều vừa đủ, không thừa cuốn nào. Biết số sách nhiều hơn $140$ cuốn và ít hơn $180$ cuốn. Hỏi thư viện quyên góp được bao nhiêu cuốn sách?
	\loigiai{
		Số sách là bội chung của $12$ và $18$.\\
		$12=2^{2}\cdot 3$, $18=2\cdot 3^{2}$, nên bội chung nhỏ nhất là $2^{2}\cdot 3^{2}=36$.\\
		Các bội của $36$ là $0;\,36;\,72;\,108;\,144;\,180;\,216;\ldots$\\
		Số sách $n$ thỏa mãn $140<n<180$, nên $n=144$.\\
		Thử lại: $144:12=12$, $144:18=8$.\\
		Vậy thư viện quyên góp được $144$ cuốn sách.
	}
\end{bt}

%% Dạng 6. Hình học thực tế — hình chữ nhật và hình thang, có chi phí.
\begin{bt}
	\immini[thm]{
		Một sân hình chữ nhật dài $16$ m và rộng $10$ m. Ở giữa sân có một bồn hoa hình thang với hai đáy $10$ m và $6$ m, chiều cao $4$ m, như hình vẽ. Phần còn lại của sân được lát gạch.
		\begin{enumerate}
			\item Tính diện tích bồn hoa.
			\item Tính diện tích phần lát gạch.
			\item Biết giá lát gạch là $125\,000$ đồng/m$^{2}$. Tính số tiền phải trả để lát phần sân còn lại.
		\end{enumerate}
	}{
		\begin{tikzpicture}[scale=0.28,line width=0.8pt,line join=round,>=stealth]
			\draw (0,0) rectangle (16,10);
			\draw (3,3)--(13,3)--(11,7)--(5,7)--cycle;
			\node at (8,5) {Bồn hoa};
			\draw[<->] (0,-1.2)--(16,-1.2) node[midway,below]{$16$ m};
			\draw[<->] (17.2,0)--(17.2,10) node[midway,right]{$10$ m};
			\draw[<->] (3,2.1)--(13,2.1) node[midway,below]{$10$ m};
			\draw[<->] (5,7.8)--(11,7.8) node[midway,above]{$6$ m};
			\draw[<->] (14.2,3)--(14.2,7) node[midway,right]{$4$ m};
		\end{tikzpicture}
	}
	\loigiai{
		\begin{enumerate}
			\item Diện tích bồn hoa là $\dfrac{(10+6)\cdot 4}{2}=32$ (m$^{2}$).
			\item Diện tích sân là $16\cdot 10=160$ (m$^{2}$).\\
			Diện tích lát gạch là $160-32=128$ (m$^{2}$).
			\item Số tiền lát gạch là $128\cdot 125\,000=16\,000\,000$ (đồng).
		\end{enumerate}
	}
\end{bt}

%% Dạng 7. Nâng cao — đặt nhân tử chung để chứng minh chia hết.
\begin{bt} Cho $B=7^{4}-7^{2}$. Chứng minh rằng $B$ chia hết cho $48$.
	\loigiai{
		$B=7^{4}-7^{2}=7^{2}\cdot(7^{2}-1)=49\cdot(49-1)=49\cdot 48$.\\
		Vì $B=49\cdot 48$ nên $B$ chia hết cho $48$.
	}
\end{bt}
