	\
	\begin{listEX}[3]

		\item $x - 12 = -27$;
		\item $(x+3)^2+2^4.11=2^2.3^2.5$.

		a) $x=10$ \hfill b) $x=-15$ \hfill c) $x=-1, x=-5$

\begin{bt} Một mảnh đất hình chữ nhật có diện tích $1500$ m$^2$, chiều rộng là $30$ m.
	\immini{
		\begin{enumerate}
			\item Tính chiều dài mảnh đất.
			\item Người ta chia mảnh đất làm bốn phần: phần dùng làm nhà ở có dạng hình chữ nhật chiều rộng $15$ m; chỗ để xe có dạng hình vuông cạnh $8$ m; phần dùng làm vườn hoa có dạng hình chữ nhật và phần còn lại dùng làm lối đi. Tính diện tích chỗ để xe, diện tích vườn hoa và diện tích lối đi.
		\end{enumerate}
}{
\begin{tikzpicture}[scale=1,line width=0.8pt,line join=round,>=stealth]
	\def\s{8}
	\path (0,0) coordinate (A)
	(8/\s,0) coordinate (B)
	(8/\s,8/\s) coordinate (M)
	(20/\s,0) coordinate (C)
	(50/\s,10/\s) coordinate (D)
	(50/\s,15/\s) coordinate (E)
	(50/\s,30/\s) coordinate (F)
	;
	\draw (A) rectangle (M) 
	(C) rectangle (D)
	(A) rectangle (E)
	(A) rectangle (F)
	(4/\s,5/\s)node{\tiny Chỗ}
	(4/\s,3/\s)node{\tiny để xe}
	(12/\s,12/\s)node{Lối đi}
	(35/\s,5/\s)node{Vườn hoa}
	(25/\s,22/\s)node{Nhà ở}
	;
	\draw[<->] (A)++(-90:0.3)--++(0:8/\s)node[midway,below]{8 m};

	\draw[<->] (F)++(0:0.3)--++(-90:15/\s)node[midway,sloped,above]{15 m};  
	\draw[<->] (F)++(0:1)--++(-90:30/\s)node[midway,right]{30 m};  
	\draw[<->] (E)++(0:0.3)--++(-90:5/\s)node[midway,sloped,above]{5 m};

}

\end{bt}
