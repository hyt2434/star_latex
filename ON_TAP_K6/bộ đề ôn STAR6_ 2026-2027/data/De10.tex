%% Dạng 1. Thực hiện phép tính — 4 ý: a cơ bản, b tính hợp lý, c và d nhiều ngoặc và lũy thừa.
\begin{bt} Thực hiện phép tính:
	\begin{listEX}
		\item $46+(-28)-13$;
		\item $(-15)\cdot 24+(-15)\cdot 16$;
		\item $96:\left[4^{2}-2^{3}\right]+(-1)^{2025}$;
		\item $(-3)^{2}\cdot 4-\left\{50-\left[2^{5}-7\cdot(-2)^{0}\right]\right\}$.
	\end{listEX}
	\loigiai{
		\begin{listEX}
			\item $46+(-28)-13=46-28-13=5$.
			\item $(-15)\cdot 24+(-15)\cdot 16=(-15)\cdot(24+16)=(-15)\cdot 40=-600$.
			\item $96:\left[4^{2}-2^{3}\right]+(-1)^{2025}$\\
			$=96:(16-8)+(-1)$\\
			Vì $2025$ là số lẻ nên $(-1)^{2025}=-1$.\\
			$=96:8+(-1)=12-1=11$.
			\item $(-3)^{2}\cdot 4-\left\{50-\left[2^{5}-7\cdot(-2)^{0}\right]\right\}$\\
			$=9\cdot 4-\left\{50-\left[32-7\cdot 1\right]\right\}$\\
			$=36-\left\{50-[32-7]\right\}$\\
			$=36-\{50-25\}$\\
			$=36-25=11$.
		\end{listEX}
	}
\end{bt}

%% Dạng 2. Tìm x — 4 ý: a cơ bản, b trị tuyệt đối, c ngoặc lồng, d có x ở cả hai vế.
\begin{bt} Tìm số nguyên $x$, biết:
	\begin{listEX}
		\item $7x=-63$;
		\item $|4x-2|=18$;
		\item $3\left\{2\left[x-(-2)\right]-6\right\}=18$;
		\item $8-3(x-2)=2(x+1)-x$.
	\end{listEX}
	\loigiai{
		\begin{listEX}
			\item $7x=-63$, suy ra $x=-63:7=-9$.
			\item $|4x-2|=18$ nên có hai trường hợp.\\
			Trường hợp 1: $4x-2=18$, nên $4x=20$, $x=5$.\\
			Trường hợp 2: $4x-2=-18$, nên $4x=-16$, $x=-4$.\\
			Vậy $x=5$ hoặc $x=-4$.
			\item $3\left\{2\left[x-(-2)\right]-6\right\}=18$\\
			$3\left[2(x+2)-6\right]=18$\\
			$3(2x+4-6)=18$\\
			$3(2x-2)=18$\\
			$6x-6=18$\\
			$6x=24$\\
			$x=4$.
			\item $8-3(x-2)=2(x+1)-x$\\
			$8-3x+6=2x+2-x$\\
			$14-3x=x+2$\\
			$12=4x$\\
			$x=3$.
		\end{listEX}
	}
\end{bt}

%% Dạng 3. Tập hợp — 2 ý, số nguyên và số tự nhiên.
\begin{bt} Biết số $0$ là số tự nhiên. Gọi $A$ là tập hợp các số nguyên $x$ thỏa mãn $-9\le x\le 9$, $x$ chia hết cho $3$ và $x$ không chia hết cho $6$. Gọi $B$ là tập hợp các số tự nhiên nhỏ hơn $20$, chia hết cho $3$ và không chia hết cho $2$.
	\begin{enumerate}
		\item Liệt kê các phần tử của $A$ và của $B$.
		\item Liệt kê các số vừa thuộc $A$ vừa thuộc $B$. Liệt kê các số thuộc $A$ nhưng không thuộc $B$.
	\end{enumerate}
	\loigiai{
		\begin{enumerate}
			\item Các số nguyên từ $-9$ đến $9$ chia hết cho $3$ là\\
			$-9;\,-6;\,-3;\,0;\,3;\,6;\,9$.\\
			Trong các số này, số không chia hết cho $6$ là $-9;\,-3;\,3;\,9$.\\
			Vậy $A=\{-9;\,-3;\,3;\,9\}$.\\
			Các số tự nhiên nhỏ hơn $20$ và chia hết cho $3$ là $0;\,3;\,6;\,9;\,12;\,15;\,18$.\\
			Trong các số này, số không chia hết cho $2$ là $3;\,9;\,15$.\\
			Vậy $B=\{3;\,9;\,15\}$.
			\item Số vừa thuộc $A$ vừa thuộc $B$ là $3$ và $9$.\\
			Các số thuộc $A$ nhưng không thuộc $B$ là $-9$ và $-3$.
		\end{enumerate}
	}
\end{bt}

%% Dạng 4. Ước và bội — thực tế cắt thanh gỗ, học sinh tự nhận ra ƯCLN.
\begin{bt} Người ta có những thanh gỗ dài $48$ cm và những thanh gỗ dài $36$ cm. Người ta cắt các thanh này thành những đoạn bằng nhau, dài nhất có thể, và không dư gỗ.
	\begin{enumerate}
		\item Mỗi đoạn dài bao nhiêu xăng-ti-mét?
		\item Mỗi thanh $48$ cm cắt được bao nhiêu đoạn? Mỗi thanh $36$ cm cắt được bao nhiêu đoạn?
	\end{enumerate}
	\loigiai{
		\begin{enumerate}
			\item Độ dài mỗi đoạn là ước chung lớn nhất của $48$ và $36$.\\
			$48=2^{4}\cdot 3$, $36=2^{2}\cdot 3^{2}$, nên ước chung lớn nhất là $2^{2}\cdot 3=12$.\\
			Mỗi đoạn dài $12$ cm.
			\item Thanh $48$ cm cắt được $48:12=4$ đoạn.\\
			Thanh $36$ cm cắt được $36:12=3$ đoạn.
		\end{enumerate}
	}
\end{bt}

%% Dạng 5. Toán thực tế — vận tốc, quãng đường, thời gian.
\begin{bt} Quãng đường từ nhà bạn Lan đến thư viện dài $2\,400$ m. Vận tốc đi bộ của bạn Lan là $80$ m/phút. Vận tốc đi xe đạp gấp $3$ lần vận tốc đi bộ.
	\begin{enumerate}
		\item Bạn Lan đi bộ hết bao nhiêu phút?
		\item Bạn Lan đi xe đạp hết bao nhiêu phút?
		\item Đi xe đạp ít hơn đi bộ bao nhiêu phút?
	\end{enumerate}
	\loigiai{
		\begin{enumerate}
			\item Thời gian đi bộ là $2\,400:80=30$ (phút).
			\item Vận tốc xe đạp là $80\cdot 3=240$ (m/phút).\\
			Thời gian đi xe đạp là $2\,400:240=10$ (phút).
			\item Đi xe đạp ít hơn đi bộ $30-10=20$ (phút).
		\end{enumerate}
	}
\end{bt}

%% Dạng 6. Hình học thực tế — lối đi bao quanh sân.
\begin{bt}
	\immini[thm]{
		Một mảnh vườn hình chữ nhật dài $25$ m, rộng $16$ m. Xung quanh vườn, sát bốn cạnh, có một lối đi rộng $2$ m. Phần bên trong lối đi dùng để trồng hoa.
		\begin{enumerate}
			\item Tính diện tích phần trồng hoa.
			\item Tính diện tích lối đi.
			\item Biết giá lát lối đi là $200\,000$ đồng/m$^{2}$. Tính số tiền lát lối đi.
		\end{enumerate}
	}{
		\begin{tikzpicture}[scale=0.22,line width=0.8pt,line join=round,>=stealth]
			\draw (0,0) rectangle (25,16);
			\draw (2,2) rectangle (23,14);
			\node at (12.5,8) {Trồng hoa};
			\draw[<->] (0,-1.4)--(25,-1.4) node[midway,below]{$25$ m};
			\draw[<->] (26.2,0)--(26.2,16) node[midway,right]{$16$ m};
			\draw[<->] (-1.2,14)--(-1.2,16) node[midway,left]{$2$ m};
		\end{tikzpicture}
	}
	\loigiai{
		\begin{enumerate}
			\item Lối đi rộng $2$ m ở cả hai đầu chiều dài và hai đầu chiều rộng.\\
			Chiều dài phần trồng hoa là $25-2-2=21$ (m).\\
			Chiều rộng phần trồng hoa là $16-2-2=12$ (m).\\
			Diện tích trồng hoa là $21\cdot 12=252$ (m$^{2}$).
			\item Diện tích cả mảnh vườn là $25\cdot 16=400$ (m$^{2}$).\\
			Diện tích lối đi là $400-252=148$ (m$^{2}$).
			\item Số tiền lát lối đi là $148\cdot 200\,000=29\,600\,000$ (đồng).
		\end{enumerate}
	}
\end{bt}

%% Dạng 7. Nâng cao — rút gọn thương của hai tổng lũy thừa cùng cơ số.
\begin{bt} Tính nhanh:
	\[
	\left(2^{7}+2^{8}+2^{9}\right):\left(2^{5}+2^{6}+2^{7}\right).
	\]
	\loigiai{
		$\left(2^{7}+2^{8}+2^{9}\right):\left(2^{5}+2^{6}+2^{7}\right)$\\
		$=\left[2^{7}\cdot(1+2+2^{2})\right]:\left[2^{5}\cdot(1+2+2^{2})\right]$\\
		$=\left[2^{7}\cdot(1+2+4)\right]:\left[2^{5}\cdot(1+2+4)\right]$\\
		$=(2^{7}\cdot 7):(2^{5}\cdot 7)$\\
		$=2^{7}:2^{5}=2^{2}=4$.
	}
\end{bt}
