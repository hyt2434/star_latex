%% Dạng 1. Thực hiện phép tính — 4 ý: a cơ bản, b tính hợp lý, c và d nhiều ngoặc và lũy thừa.
\begin{bt} Thực hiện phép tính:
	\begin{listEX}
		\item $(-35)+19+(-4)$;
		\item $36\cdot 17+36\cdot 13-36\cdot 10$;
		\item $(-2)^{3}\cdot 5+\left[48:(-6)-(-3)^{2}\right]$;
		\item $2^{6}-3\cdot\left\{20-\left[4^{2}+(-2)^{3}\cdot 3^{0}\right]\right\}$.
	\end{listEX}
	\loigiai{
		\begin{listEX}
			\item $(-35)+19+(-4)=-35+19-4=-20$.
			\item $36\cdot 17+36\cdot 13-36\cdot 10=36\cdot(17+13-10)=36\cdot 20=720$.
			\item $(-2)^{3}\cdot 5+\left[48:(-6)-(-3)^{2}\right]$\\
			$=(-8)\cdot 5+\left[-8-9\right]$\\
			$=-40+(-17)=-57$.
			\item $2^{6}-3\cdot\left\{20-\left[4^{2}+(-2)^{3}\cdot 3^{0}\right]\right\}$\\
			$=64-3\cdot\left\{20-\left[16+(-8)\cdot 1\right]\right\}$\\
			$=64-3\cdot\left\{20-[16-8]\right\}$\\
			$=64-3\cdot(20-8)$\\
			$=64-3\cdot 12=64-36=28$.
		\end{listEX}
	}
\end{bt}

%% Dạng 2. Tìm x — 4 ý: a cơ bản, b trị tuyệt đối, c ngoặc lồng, d có x ở cả hai vế.
\begin{bt} Tìm số nguyên $x$, biết:
	\begin{listEX}
		\item $-6x=42$;
		\item $|x-8|=5$;
		\item $5-2\left[4-(x+1)\right]=-7$;
		\item $6(x-1)-2(x+4)=3x-15$.
	\end{listEX}
	\loigiai{
		\begin{listEX}
			\item $-6x=42$, suy ra $x=42:(-6)=-7$.
			\item $|x-8|=5$ nên có hai trường hợp.\\
			Trường hợp 1: $x-8=5$, nên $x=13$.\\
			Trường hợp 2: $x-8=-5$, nên $x=3$.\\
			Vậy $x=13$ hoặc $x=3$.
			\item $5-2\left[4-(x+1)\right]=-7$\\
			$5-2(4-x-1)=-7$\\
			$5-2(3-x)=-7$\\
			$5-(6-2x)=-7$\\
			$5-6+2x=-7$\\
			$2x=-6$\\
			$x=-3$.
			\item $6(x-1)-2(x+4)=3x-15$\\
			$6x-6-2x-8=3x-15$\\
			$4x-14=3x-15$\\
			$x=-1$.
		\end{listEX}
	}
\end{bt}

%% Dạng 3. Tập hợp — 2 ý, số nguyên và số tự nhiên.
\begin{bt} Biết số $0$ là số tự nhiên. Gọi $P$ là tập hợp các số nguyên $x$ thỏa mãn $-20\le x<16$ và $x$ chia hết cho cả $2$ và $5$. Gọi $Q$ là tập hợp các số tự nhiên không vượt quá $18$ và chia hết cho $4$.
	\begin{enumerate}
		\item Liệt kê các phần tử của $P$ và của $Q$.
		\item Liệt kê các số vừa thuộc $P$ vừa thuộc $Q$. Liệt kê các số thuộc $Q$ nhưng không thuộc $P$.
	\end{enumerate}
	\loigiai{
		\begin{enumerate}
			\item Vì $x$ chia hết cho cả $2$ và $5$ nên $x$ chia hết cho $10$.\\
			Các số nguyên $x$ thỏa mãn $-20\le x<16$ và chia hết cho $10$ là $-20;\,-10;\,0;\,10$.\\
			Vậy $P=\{-20;\,-10;\,0;\,10\}$.\\
			Các số tự nhiên không vượt quá $18$ và chia hết cho $4$ là $0;\,4;\,8;\,12;\,16$.\\
			Vậy $Q=\{0;\,4;\,8;\,12;\,16\}$.
			\item Số vừa thuộc $P$ vừa thuộc $Q$ là $0$.\\
			Các số thuộc $Q$ nhưng không thuộc $P$ là $4;\,8;\,12;\,16$.
		\end{enumerate}
	}
\end{bt}

%% Dạng 4. Ước và bội — thực tế về lịch chạy xe, học sinh tự nhận ra BCNN.
\begin{bt} Ở một bến, xe tuyến A cứ $15$ phút xuất phát một chuyến, xe tuyến B cứ $20$ phút xuất phát một chuyến. Lúc $6$ giờ, hai tuyến cùng xuất phát.
	\begin{enumerate}
		\item Hỏi sau ít nhất bao nhiêu phút thì hai tuyến lại cùng xuất phát?
		\item Lúc đó là mấy giờ?
	\end{enumerate}
	\loigiai{
		\begin{enumerate}
			\item Hai tuyến cùng xuất phát khi số phút trôi qua là bội chung của $15$ và $20$.\\
			$15=3\cdot 5$, $20=2^{2}\cdot 5$, nên bội chung nhỏ nhất là $2^{2}\cdot 3\cdot 5=60$.\\
			Vậy sau ít nhất $60$ phút, hai tuyến lại cùng xuất phát.
			\item $6$ giờ cộng $60$ phút là $7$ giờ.
		\end{enumerate}
	}
\end{bt}

%% Dạng 5. Toán thực tế — so sánh hai cách mua.
\begin{bt} Cô Mai cần mua $6$ mét vải để may áo.
	\begin{enumerate}
		\item Quầy thứ nhất bán $45\,000$ đồng một mét. Cô Mai phải trả bao nhiêu tiền?
		\item Quầy thứ hai bán $40\,000$ đồng một mét, nhưng mỗi bộ áo phải mua thêm $2$ mét vải lót, giá $25\,000$ đồng một mét. Cô Mai phải trả bao nhiêu tiền?
		\item Quầy nào rẻ hơn và rẻ hơn bao nhiêu tiền?
	\end{enumerate}
	\loigiai{
		\begin{enumerate}
			\item Tiền vải ở quầy thứ nhất là $6\cdot 45\,000=270\,000$ (đồng).
			\item Tiền vải chính là $6\cdot 40\,000=240\,000$ (đồng).\\
			Tiền vải lót là $2\cdot 25\,000=50\,000$ (đồng).\\
			Tổng tiền là $240\,000+50\,000=290\,000$ (đồng).
			\item $290\,000-270\,000=20\,000$ (đồng).\\
			Vậy quầy thứ nhất rẻ hơn $20\,000$ đồng.
		\end{enumerate}
	}
\end{bt}

%% Dạng 6. Hình học thực tế — hình thang có một sân bên trong.
\begin{bt}
	\immini[thm]{
		Một mảnh vườn hình thang có hai đáy $18$ m và $10$ m, chiều cao $8$ m. Giữa vườn có một sân phơi hình chữ nhật dài $6$ m, rộng $4$ m. Phần còn lại dùng để trồng cây.
		\begin{enumerate}
			\item Tính diện tích mảnh vườn.
			\item Tính diện tích đất trồng cây.
			\item Cứ mỗi mét vuông cần bón $2$ kg phân, giá phân là $8\,000$ đồng/kg. Tính số tiền mua phân để bón hết phần đất trồng cây.
		\end{enumerate}
	}{
		\begin{tikzpicture}[scale=0.28,line width=0.8pt,line join=round,>=stealth]
			\draw (0,0)--(18,0)--(14,8)--(4,8)--cycle;
			\draw (6,2) rectangle (12,6);
			\node at (9,4) {\footnotesize Sân};
			\draw[<->] (0,-1.1)--(18,-1.1) node[midway,below]{$18$ m};
			\draw[<->] (4,8.7)--(14,8.7) node[midway,above]{$10$ m};
			\draw[<->] (18.8,0)--(18.8,8) node[midway,right]{$8$ m};
			\node at (9,1) {\footnotesize $6$ m};
			\node at (12.9,4) {\footnotesize $4$ m};
		\end{tikzpicture}
	}
	\loigiai{
		\begin{enumerate}
			\item Diện tích mảnh vườn là $\dfrac{(18+10)\cdot 8}{2}=112$ (m$^{2}$).
			\item Diện tích sân phơi là $6\cdot 4=24$ (m$^{2}$).\\
			Diện tích đất trồng cây là $112-24=88$ (m$^{2}$).
			\item Số phân cần bón là $88\cdot 2=176$ (kg).\\
			Số tiền mua phân là $176\cdot 8\,000=1\,408\,000$ (đồng).
		\end{enumerate}
	}
\end{bt}

%% Dạng 7. Nâng cao — so sánh lũy thừa bằng cách đưa về cùng cơ số.
\begin{bt} So sánh $3^{12}$ và $9^{5}$.
	\loigiai{
		$9^{5}=(3^{2})^{5}=3^{2\cdot 5}=3^{10}$.\\
		Vì cơ số $3>1$ và $12>10$ nên $3^{12}>3^{10}$.\\
		Vậy $3^{12}>9^{5}$.
	}
\end{bt}
