\begin{bt}Cho $A=6^1+6^2+6^3+\ldots+6^{18}+6^{19}+6^{20}$. Chứng tỏ $A$ là bội của $222$.
	\loigiai{
		Ta có:
		$$
		\begin{aligned}
			& A=6^1+6^2+\ldots+6^{19}+6^{20}=\left(6^1+6^2+6^3+6^4\right)+\left(6^5+6^6+6^7+6^8\right)+\ldots+\left(6^{17}+6^{18}+6^{19}+6^{20}\right) \\
			& =1554+6^4 \cdot\left(6^1+6^2+6^3+6^4\right)+\ldots+6^{16} \cdot\left(6^1+6^2+6^3+6^4\right) \\
			& =1554+6^1 \cdot 1554+\ldots+6^{16} \cdot 1554\\
			& =1554\cdot(1+6^1+\ldots+6^{16})
		\end{aligned}
		$$
		Vì $1554\, \vdots \, 222$ nên $A \, \vdots \,222$. 
		Suy ra $A$ là bội của 222.
		
	}
\end{bt}
