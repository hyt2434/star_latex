%% Dạng 1. Thực hiện phép tính — 4 ý: a cơ bản, b tính hợp lý, c và d nhiều ngoặc và lũy thừa.
\begin{bt} Thực hiện phép tính:
	\begin{listEX}
		\item $(-42)-(-17)+(-8)$;
		\item $25\cdot 19+25\cdot 21-50\cdot 10$;
		\item $\left[(-72):8+5\cdot(-2)\right]\cdot(-1)^{4}-3^{2}$;
		\item $7^{2}\cdot 2-4\cdot\left[3^{3}-\left(5^{2}-2^{4}\right)\cdot 2^{0}\right]$.
	\end{listEX}
	\loigiai{a) $-33$ \hfill b) $500$ \hfill c) $-28$ \hfill d) $26$.}
\end{bt}

%% Dạng 2. Tìm x — 4 ý: a cơ bản, b trị tuyệt đối, c ngoặc lồng, d có x ở cả hai vế.
\begin{bt} Tìm số nguyên $x$, biết:
	\begin{listEX}
		\item $-9x=-72$;
		\item $|2x+4|=12$;
		\item $4\left[5-2(x-3)\right]=12$;
		\item $7(x+2)-4(x-1)=2(x+8)+3$.
	\end{listEX}
	\loigiai{a) $x=8$ \hfill b) $x=4$ hoặc $x=-8$ \hfill c) $x=4$ \hfill d) $x=1$.}
\end{bt}

%% Dạng 3. Tập hợp — 2 ý, số nguyên và số tự nhiên.
\begin{bt} Biết số $0$ là số tự nhiên. Gọi $E$ là tập hợp các số nguyên $x$ thỏa mãn $-18<x\le 18$, $x$ chia hết cho $6$ và $x$ không chia hết cho $4$. Gọi $F$ là tập hợp các số tự nhiên nhỏ hơn $24$ và chia hết cho $6$.
	\begin{enumerate}
		\item Liệt kê các phần tử của $E$ và của $F$.
		\item Liệt kê các số vừa thuộc $E$ vừa thuộc $F$. Liệt kê các số thuộc $F$ nhưng không thuộc $E$.
	\end{enumerate}
	\loigiai{a) $E=\{-6;\,6;\,18\}$, $F=\{0;\,6;\,12;\,18\}$ \hfill b) Phần chung: $\{6;\,18\}$; thuộc $F$ nhưng không thuộc $E$: $\{0;\,12\}$.}
\end{bt}

%% Dạng 4. Ước và bội — thực tế xếp hàng, học sinh tự nhận ra BCNN.
\begin{bt} Học sinh một khối lớp khi xếp thành hàng $16$ em, hàng $24$ em hoặc hàng $32$ em đều vừa đủ hàng, không thừa em nào. Biết số học sinh nhiều hơn $90$ và ít hơn $140$. Hỏi khối lớp đó có bao nhiêu học sinh?
	\loigiai{$96$ học sinh.}
\end{bt}

%% Dạng 5. Toán thực tế — so sánh hai phương án thuê.
\begin{bt} Một gia đình cần thuê máy phát điện trong một ngày. Gói thứ nhất giá $150\,000$ đồng cho cả ngày. Gói thứ hai giá $40\,000$ đồng mỗi giờ.
	\begin{enumerate}
		\item Nếu dùng máy $5$ giờ thì mỗi gói hết bao nhiêu tiền? Nên chọn gói nào, và rẻ hơn bao nhiêu?
		\item Nếu dùng máy $3$ giờ thì nên chọn gói nào, và rẻ hơn bao nhiêu?
	\end{enumerate}
	\loigiai{a) Gói thứ nhất $150\,000$ đồng, gói thứ hai $200\,000$ đồng. Chọn gói thứ nhất, rẻ hơn $50\,000$ đồng. \hfill b) Chọn gói thứ hai, rẻ hơn $30\,000$ đồng.}
\end{bt}

%% Dạng 6. Hình học thực tế — chữ nhật ghép tam giác, sơn hai mặt.
\begin{bt}
	\immini[thm]{
		Một biển quảng cáo gồm một hình chữ nhật rộng $12$ dm, cao $5$ dm, và một hình tam giác có đáy trùng cạnh $12$ dm phía trên hình chữ nhật. Chiều cao của hình tam giác kẻ vuông góc với đáy là $4$ dm.
		\begin{enumerate}
			\item Tính diện tích phần hình chữ nhật.
			\item Tính diện tích phần hình tam giác.
			\item Người ta sơn cả hai mặt của biển. Biết mỗi đề-xi-mét vuông hết $15\,000$ đồng tiền sơn. Tính số tiền sơn cả biển.
		\end{enumerate}
	}{
		\begin{tikzpicture}[scale=0.45,line width=0.8pt,line join=round,>=stealth]
			\draw (0,0) rectangle (12,5);
			\draw (0,5)--(6,9)--(12,5);
			\draw[dashed] (6,5)--(6,9);
			\node at (6,2.5) {Chữ nhật};
			\draw[<->] (0,-0.7)--(12,-0.7) node[midway,below]{$12$ dm};
			\draw[<->] (12.6,0)--(12.6,5) node[midway,right]{$5$ dm};
			\draw[<->] (6.4,5)--(6.4,9) node[midway,right]{$4$ dm};
		\end{tikzpicture}
	}
	\loigiai{a) $60$ dm$^{2}$ \hfill b) $24$ dm$^{2}$ \hfill c) $2\,520\,000$ đồng.}
\end{bt}

%% Dạng 7. Nâng cao — chứng minh chia hết bằng cách đặt nhân tử chung.
\begin{bt} Chứng minh rằng $4^{3}+4^{4}+4^{5}$ chia hết cho $21$.
	\loigiai{Hướng dẫn: $4^{3}+4^{4}+4^{5}=4^{3}\cdot(1+4+16)=64\cdot 21$, nên tổng này chia hết cho $21$.}
\end{bt}
