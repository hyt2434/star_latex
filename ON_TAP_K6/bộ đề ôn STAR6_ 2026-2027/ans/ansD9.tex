\begin{Solbook}{3}
 \par \noindent  \begin {enumerate} \item Vì $x$ chia hết cho cả $2$ và $5$ nên $x$ chia hết cho $10$.\\ Các số nguyên $x$ thỏa mãn $-20\le x<16$ và chia hết cho $10$ là $-20;\,-10;\,0;\,10$.\\ Vậy $P=\{-20;\,-10;\,0;\,10\}$.\\ Các số tự nhiên không vượt quá $18$ và chia hết cho $4$ là $0;\,4;\,8;\,12;\,16$.\\ Vậy $Q=\{0;\,4;\,8;\,12;\,16\}$. \item Số vừa thuộc $P$ vừa thuộc $Q$ là $0$.\\ Các số thuộc $Q$ nhưng không thuộc $P$ là $4;\,8;\,12;\,16$. \end {enumerate}  \hfill \qedEX 
\end{Solbook}
\begin{Solbook}{4}
 \par \noindent  \begin {enumerate} \item Hai tuyến cùng xuất phát khi số phút trôi qua là bội chung của $15$ và $20$.\\ $15=3\cdot 5$, $20=2^{2}\cdot 5$, nên bội chung nhỏ nhất là $2^{2}\cdot 3\cdot 5=60$.\\ Vậy sau ít nhất $60$ phút, hai tuyến lại cùng xuất phát. \item $6$ giờ cộng $60$ phút là $7$ giờ. \end {enumerate}  \hfill \qedEX 
\end{Solbook}
