 \par \noindent  \begin {enumerate} \item Ta có \begin {align*} (3x-2)^2-(x+4)^2&=0\\ \big [(3x-2)-(x+4)\big ]\big [(3x-2)+(x+4)\big ]&=0\\ (2x-6)(4x+2)&=0. \end {align*} Do đó $x=3$ hoặc $x=-\dfrac {1}{2}$. \par \item Nhân cả hai vế với $12>0$, ta được \begin {align*} 3(3x-2)-4(x+5)&<2(5x+1)\\ 5x-26&<10x+2\\ -28&<5x\\ x&>-\dfrac {28}{5}. \end {align*} Vậy tập nghiệm là $S=\left (-\dfrac {28}{5};+\infty \right )$. \par \item Quy đồng và rút gọn từng phương trình, hệ đã cho tương đương với \[ \begin {cases} 7x+4y=10,\\ 3x-2y=8. \end {cases} \] Nhân phương trình thứ hai với $2$ rồi cộng với phương trình thứ nhất: \[ 13x=26\Leftrightarrow x=2. \] Thay $x=2$ vào $3x-2y=8$, ta được $y=-1$. Vậy hệ có nghiệm $(x;y)=(2;-1)$. \end {enumerate}  \hfill \qedEX
