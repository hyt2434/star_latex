\begin{bt}[1,5 điểm]
Giải các phương trình, bất phương trình và hệ phương trình sau:
\begin{listEX}[3]
\item $(2x-1)^2=9(x-2)^2$
\item $3-\dfrac{2x-5}{4}\le\dfrac{3x+1}{6}-\dfrac{x-2}{3}$
\item $\begin{cases}
\dfrac{3x-y}{4}+\dfrac{x+2y}{3}=\dfrac{1}{6}\\
2(x+1)-3(y-2)=-3
\end{cases}$
\end{listEX}


\loigiai{
\begin{enumerate}
\item Ta có
\begin{align*}
&(2x-1)^2=9(x-2)^2\\
&(2x-1)^2-\big[3(x-2)\big]^2=0\\
&\big[(2x-1)-3(x-2)\big]\big[(2x-1)+3(x-2)\big]=0\\
&(5-x)(5x-7)=0.
\end{align*}
Do đó $x=5$ hoặc $x=\dfrac{7}{5}$.

\item Thu gọn vế phải rồi nhân cả hai vế với $12>0$:
\begin{align*}
3-\dfrac{2x-5}{4}&\le \dfrac{x+5}{6}\\
3(17-2x)&\le 2(x+5)\\
51-6x&\le 2x+10\\
41&\le 8x\\
x&\ge \dfrac{41}{8}.
\end{align*}
Vậy bất phương trình có nghiệm $x\ge\dfrac{41}{8}$.

\item Dùng phương pháp cộng đại số, ta có
\begin{align*}
&\begin{cases}
\dfrac{3x-y}{4}+\dfrac{x+2y}{3}=\dfrac{1}{6},\\
2(x+1)-3(y-2)=-3
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
13x+5y=2,\\
2x-3y=-11
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
39x+15y=6,\\
10x-15y=-55
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
49x=-49,\\
2x-3y=-11
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
x=-1,\\
y=3.
\end{cases}
\end{align*}
Vậy hệ phương trình có nghiệm $(x;y)=(-1;3)$.
\end{enumerate}
}
\end{bt}
