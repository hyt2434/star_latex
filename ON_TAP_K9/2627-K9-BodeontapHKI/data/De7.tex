%Bài 1:
\begin{bt}[1,5 điểm]
Giải các phương trình, bất phương trình và hệ phương trình sau:
\begin{listEX}[3]
\item $(3x-2)^2-(x+4)^2=0$
\item $\dfrac{3x-2}{4}-\dfrac{x+5}{3}<\dfrac{5x+1}{6}$
\item $\begin{cases}
\dfrac{2x-y}{3}+\dfrac{x+2y}{2}=\dfrac{5}{3}\\
3(x-1)-2(y+2)=1
\end{cases}$
\end{listEX}
\loigiai{
\begin{enumerate}
\item Ta có
\begin{align*}
(3x-2)^2-(x+4)^2&=0\\
\big[(3x-2)-(x+4)\big]\big[(3x-2)+(x+4)\big]&=0\\
(2x-6)(4x+2)&=0.
\end{align*}
Do đó $x=3$ hoặc $x=-\dfrac{1}{2}$.

\item Nhân cả hai vế với $12>0$, ta được
\begin{align*}
3(3x-2)-4(x+5)&<2(5x+1)\\
5x-26&<10x+2\\
-28&<5x\\
x&>-\dfrac{28}{5}.
\end{align*}
Vậy bất phương trình có nghiệm $x>-\dfrac{28}{5}$.

\item Dùng phương pháp cộng đại số, ta có
\begin{align*}
&\begin{cases}
\dfrac{2x-y}{3}+\dfrac{x+2y}{2}=\dfrac{5}{3},\\
3(x-1)-2(y+2)=1
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
7x+4y=10,\\
3x-2y=8
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
7x+4y=10,\\
6x-4y=16
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
13x=26,\\
3x-2y=8
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
x=2,\\
y=-1.
\end{cases}
\end{align*}
Vậy hệ phương trình có nghiệm $(x;y)=(2;-1)$.
\end{enumerate}
}
\end{bt}

\begin{bt}[1 điểm]
Rút gọn các biểu thức sau:
\begin{listEX}[2]
\item $\sqrt{23+8\sqrt{7}}+\sqrt{23-8\sqrt{7}}-\sqrt{14-4\sqrt{10}}$
\item $\dfrac{\sqrt{80}-\sqrt{5}}{\sqrt{5}-\sqrt{3}}+\dfrac{3\sqrt{3}}{\sqrt{5}+\sqrt{3}}$
\end{listEX}
\loigiai{
\begin{enumerate}
\setlength{\itemsep}{1em}

\item Ta có
\begin{align*}
23+8\sqrt{7}
&=16+2\cdot4\sqrt{7}+7\\
&=\left(4+\sqrt{7}\right)^2,\\
23-8\sqrt{7}
&=16-2\cdot4\sqrt{7}+7\\
&=\left(4-\sqrt{7}\right)^2,\\
14-4\sqrt{10}
&=10-2\cdot\sqrt{10}\cdot2+4\\
&=\left(\sqrt{10}-2\right)^2.
\end{align*}
Vì $4-\sqrt{7}>0$ và $\sqrt{10}-2>0$ nên
\begin{align*}
\sqrt{23+8\sqrt{7}}+\sqrt{23-8\sqrt{7}}
-\sqrt{14-4\sqrt{10}}\\
&=4+\sqrt{7}+4-\sqrt{7}-\left(\sqrt{10}-2\right)\\
&=10-\sqrt{10}.
\end{align*}
Vậy biểu thức đã cho rút gọn thành $10-\sqrt{10}$.

\item Ta có $\sqrt{80}=4\sqrt{5}$, do đó
\begin{align*}
\dfrac{\sqrt{80}-\sqrt{5}}{\sqrt{5}-\sqrt{3}}
+\dfrac{3\sqrt{3}}{\sqrt{5}+\sqrt{3}}\\
&=\dfrac{3\sqrt{5}}{\sqrt{5}-\sqrt{3}}
+\dfrac{3\sqrt{3}}{\sqrt{5}+\sqrt{3}}\\
&=\dfrac{3\sqrt{5}\left(\sqrt{5}+\sqrt{3}\right)}{5-3}
+\dfrac{3\sqrt{3}\left(\sqrt{5}-\sqrt{3}\right)}{5-3}\\
&=\dfrac{15+3\sqrt{15}+3\sqrt{15}-9}{2}\\
&=3+3\sqrt{15}.
\end{align*}
Vậy biểu thức đã cho rút gọn thành $3+3\sqrt{15}$.
\end{enumerate}
}
\end{bt}

%Bài 3
\begin{bt}[1,0 điểm]
Cô Hà mua dụng cụ học tập tại một cửa hàng. Lần thứ nhất, cô mua $5$ cây bút và $3$ quyển vở hết $70\,000$ đồng. Lần thứ hai, cô mua $2$ cây bút và $4$ quyển vở hết $56\,000$ đồng. Biết các cây bút cùng giá và các quyển vở cùng giá. Tính giá tiền của một cây bút và một quyển vở.
\loigiai{
Gọi $x$ (nghìn đồng) là giá một cây bút và $y$ (nghìn đồng) là giá một quyển vở, với $x>0$, $y>0$.

Lần mua thứ nhất cho ta
\[
5x+3y=70. \tag{1}
\]
Lần mua thứ hai cho ta
\[
2x+4y=56. \tag{2}
\]
Chia hai vế của $(2)$ cho $2$:
\[
x+2y=28 \Rightarrow x=28-2y. \tag{3}
\]
Thế $(3)$ vào $(1)$:
\begin{align*}
5(28-2y)+3y&=70\\
140-10y+3y&=70\\
140-7y&=70\\
-7y&=-70\\
y&=10.
\end{align*}
Thay $y=10$ vào $(3)$:
\[
x=28-2\cdot 10=8.
\]
Cặp $(x;y)=(8;10)$ thỏa $x>0$, $y>0$.

\emph{Kiểm tra.}
\(5\cdot 8+3\cdot 10=70\) và \(2\cdot 8+4\cdot 10=56\), đúng với giả thiết.

Vậy một cây bút giá $8\,000$ đồng và một quyển vở giá $10\,000$ đồng.
}
\end{bt}

\begin{bt}[1,0 điểm]
\immini[thm]
{Một khung ảnh hình chữ nhật có chiều dài $18$ cm và chiều rộng $12$ cm. Phần viền xung quanh có bề rộng đều là $x$ cm ($0<x<6$), như hình vẽ.
\begin{enumerate}
\item Viết biểu thức diện tích hình chữ nhật lớn, diện tích phần đặt ảnh và diện tích phần viền theo $x$.
\item Biết diện tích phần viền là $104$ cm$^2$. Tìm $x$.
\end{enumerate}
}
{
\begin{tikzpicture}[scale=0.22, font=\footnotesize, line join=round]
	\draw[fill=gray!30] (0,0) rectangle (18,12);
	\draw[fill=white] (2.4,2.4) rectangle (15.6,9.6);
	\draw[<->] (0,-1.6)--(18,-1.6);
	\node[below] at (9,-1.6) {$18$ cm};
	\draw[<->] (-1.8,0)--(-1.8,12);
	\node[left] at (-1.8,6) {$12$ cm};
	\draw[<->] (15.6,6)--(18,6);
	\node[above] at (16.8,6) {$x$};
\end{tikzpicture}
}
\loigiai{
\begin{enumerate}
\item Diện tích hình chữ nhật lớn là
\[
S_{\text{lớn}}=18\cdot 12=216\ \text{(cm}^2\text{)}.
\]
Mỗi chiều của phần đặt ảnh ngắn hơn khung ngoài $2x$ cm, nên phần đặt ảnh có chiều dài $18-2x$ (cm) và chiều rộng $12-2x$ (cm). Diện tích phần đặt ảnh là
\begin{align*}
S_{\text{ảnh}}
&=(18-2x)(12-2x)\\
&=216-36x-24x+4x^2\\
&=4x^2-60x+216\ \text{(cm}^2\text{)}.
\end{align*}
Diện tích phần viền là
\begin{align*}
S_{\text{viền}}
&=S_{\text{lớn}}-S_{\text{ảnh}}\\
&=216-(4x^2-60x+216)\\
&=60x-4x^2\\
&=4x(15-x)\ \text{(cm}^2\text{)}.
\end{align*}

\item Ta có $4x(15-x)=104$. Vì $x\neq 0$ và $4\neq 0$, chia hai vế cho $4$:
\begin{align*}
x(15-x)&=26\\
15x-x^2&=26\\
x^2-15x+26&=0\\
(x-2)(x-13)&=0.
\end{align*}
Suy ra $x=2$ hoặc $x=13$.

Do $0<x<6$ nên $x=13$ bị loại. Vậy $x=2$ cm.

\emph{Kiểm tra.} Phần đặt ảnh có kích thước $14$ cm và $8$ cm, diện tích $112$ cm$^2$. Phần viền có diện tích $216-112=104$ cm$^2$, đúng với giả thiết.
\end{enumerate}
}
\end{bt}

\begin{bt}[1,0 điểm]
\immini[thm]
{Một tấm gỗ hình tròn bán kính $16$ cm được gắn chính giữa một gương tròn bán kính $10$ cm. Phần gỗ còn nhìn thấy là một hình vành khuyên (hình vẽ). Lấy $\pi=3{,}14$.
\begin{enumerate}
\item Tính diện tích phần gỗ còn nhìn thấy (làm tròn kết quả đến hàng phần mười centimét vuông).
\item Người ta dán một đường kim tuyến quanh mép ngoài của tấm gỗ. Tính độ dài đường kim tuyến đó (làm tròn kết quả đến hàng phần mười centimét).
\end{enumerate}
}
{
\begin{tikzpicture}[scale=0.11, font=\scriptsize, line join=round]
	\fill[brown!40] (0,0) circle (16);
	\fill[cyan!20] (0,0) circle (10);
	\draw (0,0) circle (16);
	\draw (0,0) circle (10);
	\draw (0,0)--(16,0);
	\node[above] at (8,0) {$16$};
	\draw (0,0)--(125:10);
	\node[above right] at (145:5) {$10$};
	\fill (0,0) circle (2pt) node[below right] {$O$};
\end{tikzpicture}
}
\loigiai{
\begin{enumerate}
\item Phần gỗ còn nhìn thấy là hình vành khuyên có bán kính ngoài $R=16$ cm và bán kính trong $r=10$ cm. Diện tích phần này là
\begin{align*}
S&=\pi R^2-\pi r^2\\
&=\pi\left(16^2-10^2\right)\\
&=3{,}14(256-100)\\
&=3{,}14\cdot 156\\
&=489{,}84\ \text{(cm}^2\text{)}.
\end{align*}
Làm tròn đến hàng phần mười: $S\approx 489{,}8$ cm$^2$.

Vậy diện tích phần gỗ còn nhìn thấy khoảng $489{,}8$ cm$^2$.

\item Đường kim tuyến là chu vi đường tròn mép ngoài, bán kính $16$ cm:
\begin{align*}
C&=2\pi R\\
&=2\cdot 3{,}14\cdot 16\\
&=100{,}48\ \text{(cm)}.
\end{align*}
Làm tròn đến hàng phần mười: $C\approx 100{,}5$ cm.

Vậy độ dài đường kim tuyến khoảng $100{,}5$ cm.
\end{enumerate}
}
\end{bt}
