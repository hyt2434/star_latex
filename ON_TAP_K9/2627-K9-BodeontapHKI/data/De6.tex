\begin{bt}[1,5 điểm]
Giải các phương trình, bất phương trình và hệ phương trình sau:
\begin{listEX}[3]
\item $\dfrac{3x+5}{x-2}-\dfrac{3x-1}{x+2}=\dfrac{26}{x^2-4}$
\item $\dfrac{2x-3}{3}-\dfrac{5x+1}{4}\ge\dfrac{1-2x}{6}$
\item $\begin{cases}
2(x-1)-3(y+2)=-11\\
\dfrac{x+2y}{3}+\dfrac{x-y}{2}=3
\end{cases}$
\end{listEX}

\loigiai{
\begin{enumerate}
\item Điều kiện xác định: $x\ne 2$, $x\ne -2$.
\begin{align*}
\dfrac{3x+5}{x-2}-\dfrac{3x-1}{x+2}&=\dfrac{26}{x^2-4}\\
(3x+5)(x+2)-(3x-1)(x-2)&=26\\
18x+8&=26\\
x&=1.
\end{align*}
Giá trị $x=1$ thỏa mãn điều kiện xác định. Vậy phương trình có nghiệm $x=1$.

\item Nhân cả hai vế với $12>0$, ta được
\begin{align*}
4(2x-3)-3(5x+1)&\ge 2(1-2x)\\
8x-12-15x-3&\ge 2-4x\\
-3x&\ge 17\\
x&\le -\dfrac{17}{3}.
\end{align*}
Vậy bất phương trình có nghiệm $x\le -\dfrac{17}{3}$.

\item Dùng phương pháp thế, ta có
\begin{align*}
&\begin{cases}
2(x-1)-3(y+2)=-11,\\
\dfrac{x+2y}{3}+\dfrac{x-y}{2}=3
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
2x-3y=-3,\\
5x+y=18
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
2x-3(18-5x)=-3,\\
y=18-5x
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
17x=51,\\
y=18-5x
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
x=3,\\
y=3.
\end{cases}
\end{align*}
Vậy hệ phương trình có nghiệm $(x;y)=(3;3)$.
\end{enumerate}
}
\end{bt}



%%Bài 2:
\begin{bt}[1 điểm]
Rút gọn các biểu thức sau:
\begin{listEX}[2]
\item $\sqrt{14+6\sqrt{5}}-\sqrt{17-12\sqrt{2}}$
\item $\dfrac{\sqrt{45}+\sqrt{5}}{\sqrt{5}-\sqrt{3}}
-\dfrac{4\sqrt{3}}{\sqrt{5}+\sqrt{3}}$
\end{listEX}
\loigiai{
\begin{enumerate}
\setlength{\itemsep}{1em}

\item Ta có
\begin{align*}
14+6\sqrt{5}
&=9+2\cdot3\sqrt{5}+5\\
&=\left(3+\sqrt{5}\right)^2,\\
17-12\sqrt{2}
&=9-2\cdot3\cdot2\sqrt{2}+8\\
&=\left(3-2\sqrt{2}\right)^2.
\end{align*}
Vì $3-2\sqrt{2}>0$ nên
\begin{align*}
\sqrt{14+6\sqrt{5}}-\sqrt{17-12\sqrt{2}}
&=\sqrt{\left(3+\sqrt{5}\right)^2}
-\sqrt{\left(3-2\sqrt{2}\right)^2}\\
&=3+\sqrt{5}-\left(3-2\sqrt{2}\right)\\
&=\sqrt{5}+2\sqrt{2}.
\end{align*}
Vậy biểu thức đã cho rút gọn thành $\sqrt{5}+2\sqrt{2}$.

\item Ta có $\sqrt{45}=3\sqrt{5}$, do đó
\begin{align*}
\dfrac{\sqrt{45}+\sqrt{5}}{\sqrt{5}-\sqrt{3}}
-\dfrac{4\sqrt{3}}{\sqrt{5}+\sqrt{3}}\\
&=\dfrac{4\sqrt{5}}{\sqrt{5}-\sqrt{3}}
-\dfrac{4\sqrt{3}}{\sqrt{5}+\sqrt{3}}\\
&=\dfrac{4\sqrt{5}\left(\sqrt{5}+\sqrt{3}\right)}{5-3}
-\dfrac{4\sqrt{3}\left(\sqrt{5}-\sqrt{3}\right)}{5-3}\\
&=10+2\sqrt{15}-\left(2\sqrt{15}-6\right)\\
&=16.
\end{align*}
Vậy biểu thức đã cho rút gọn thành $16$.
\end{enumerate}
}
\end{bt}

%%Bài 3:
\begin{bt}[1 điểm]
Một công ty du lịch thông báo giá thuê xe cho chuyến đi một chiều như sau:
\begin{itemize}
\item Hãng A: giá mở cửa $200$ nghìn đồng và thêm $4$ nghìn đồng cho mỗi kilômét.
\item Hãng B: giá mở cửa $80$ nghìn đồng và thêm $6$ nghìn đồng cho mỗi kilômét.
\end{itemize}
Gọi $x$ (km), với $x>0$, là quãng đường chuyến đi. Gọi $f(x)$ và $g(x)$ (nghìn đồng) lần lượt là số tiền phải trả cho hãng A và hãng B.
\begin{enumerate}
\item Viết công thức của $f(x)$ và $g(x)$.
\item Tìm quãng đường để số tiền thuê hai hãng bằng nhau. Nếu quãng đường dài hơn mức vừa tìm thì nên chọn hãng nào để tiết kiệm chi phí? Vì sao?
\end{enumerate}
\loigiai{
\begin{enumerate}
\item Vì hãng A thu $200$ nghìn đồng tiền mở cửa và $4$ nghìn đồng cho mỗi kilômét nên
\[
f(x)=4x+200.
\]
Vì hãng B thu $80$ nghìn đồng tiền mở cửa và $6$ nghìn đồng cho mỗi kilômét nên
\[
g(x)=6x+80.
\]

\item Hai hãng có số tiền thuê bằng nhau khi $f(x)=g(x)$. Ta có
\begin{align*}
4x+200&=6x+80\\
200-80&=6x-4x\\
120&=2x\\
x&=60.
\end{align*}
Vậy hai hãng thu cùng số tiền khi quãng đường dài $60$ km.

Xét hiệu
\begin{align*}
f(x)-g(x)
&=(4x+200)-(6x+80)\\
&=120-2x\\
&=2(60-x).
\end{align*}
Với $x>60$ thì $60-x<0$, nên $f(x)-g(x)<0$, tức là $f(x)<g(x)$.

Do đó, nếu quãng đường dài hơn $60$ km thì nên chọn hãng A để tiết kiệm chi phí.
\end{enumerate}
}
\end{bt}

%%Bài 4:
\begin{bt}[1 điểm]
Mẹ Hoa nhờ bạn mua một nồi cơm điện và một bàn là. Giá niêm yết của hai món cộng lại là $750\,000$ đồng. Hôm nay nồi cơm điện được giảm $20\%$, bàn là được giảm $10\%$ so với giá niêm yết, nên số tiền phải trả ít hơn tổng giá niêm yết $120\,000$ đồng.
\begin{enumerate}
\item Tìm giá niêm yết của mỗi món.
\item Tính số tiền thực tế phải trả cho mỗi món sau khi giảm giá.
\end{enumerate}
\loigiai{
\begin{enumerate}
\item Gọi $x$ (nghìn đồng) là giá niêm yết của nồi cơm điện và $y$ (nghìn đồng) là giá niêm yết của bàn là, với $x>0$, $y>0$.

Tổng giá niêm yết là $750$ nghìn đồng, nên
\[
x+y=750. \tag{1}
\]
Số tiền được giảm là $20\%$ giá nồi cộng với $10\%$ giá bàn là và bằng $120$ nghìn đồng, nên
\[
\dfrac{20}{100}x+\dfrac{10}{100}y=120,
\]
tức là
\[
0{,}2x+0{,}1y=120.
\]
Nhân hai vế với $10$, ta được
\[
2x+y=1200. \tag{2}
\]
Lấy $(2)$ trừ $(1)$:
\begin{align*}
(2x+y)-(x+y)&=1200-750\\
x&=450.
\end{align*}
Thay $x=450$ vào $(1)$:
\[
450+y=750 \Rightarrow y=300.
\]
Cặp $(x;y)=(450;300)$ thỏa điều kiện $x>0$, $y>0$.

Vậy giá niêm yết của nồi cơm điện là $450\,000$ đồng và giá niêm yết của bàn là là $300\,000$ đồng.

\item Sau khi giảm giá, số tiền thực tế là
\begin{align*}
\text{nồi cơm điện: }&450\,000\cdot(100\%-20\%)=450\,000\cdot 0{,}8=360\,000 \text{ (đồng)},\\
\text{bàn là: }&300\,000\cdot(100\%-10\%)=300\,000\cdot 0{,}9=270\,000 \text{ (đồng)}.
\end{align*}
Vậy phải trả $360\,000$ đồng cho nồi cơm điện và $270\,000$ đồng cho bàn là.

\emph{Kiểm tra.} Số tiền giảm là $90\,000+30\,000=120\,000$ (đồng), đúng với giả thiết.
\end{enumerate}
}
\end{bt}

\begin{bt}[1 điểm]
\immini[thm]
{Một người đứng trên mặt đất, cách gốc một cột đèn $12$ m. Mắt người cách mặt đất $1{,}5$ m. Tia nhìn từ mắt đến đỉnh cột đèn tạo với phương nằm ngang một góc $30^\circ$ (hình vẽ).
\begin{enumerate}
\item Tính chiều cao của cột đèn (làm tròn kết quả đến hàng phần mười mét).
\item Một học sinh có mắt cách mặt đất $1{,}5$ m muốn nhìn đỉnh cột đèn dưới góc nâng đúng $45^\circ$. Hỏi học sinh đó phải đứng cách gốc cột đèn bao nhiêu mét? \textit{(tính chính xác, không dùng kết quả đã làm tròn ở câu a)}
\end{enumerate}
}
{
\begin{tikzpicture}[scale=0.34, font=\footnotesize, >=stealth, line join=round]
	\pgfmathsetmacro{\hB}{1.5+8*sin(60)}
	\coordinate (N) at (0,0);
	\coordinate (M) at (0,1.5);
	\coordinate (A) at (12,0);
	\coordinate (H) at (12,1.5);
	\coordinate (B) at (12,\hB);
	\draw[thick] (N)--(A)--(B);
	\draw[dashed] (M)--(N) (M)--(H);
	\draw[thick] (M)--(B);
	\draw (2.2,1.5) arc[start angle=0, end angle=30, radius=2.2];
	\node at (3.15,2.15) {$30^\circ$};
	\node[below] at ($(N)!0.5!(A)$) {$12$ m};
	\node[left] at ($(N)!0.5!(M)$) {$1{,}5$ m};
	\draw ($(H)+(-0.45,0)$)--++(0,0.45)--++(0.45,0);
	\foreach \p/\pos in {M/left,A/below right,B/above,H/right}
		\fill (\p) circle (1.6pt) node[\pos] {$\p$};
\end{tikzpicture}
}
\loigiai{
\begin{enumerate}
\item Gọi $M$ là vị trí mắt, $A$ là gốc cột đèn, $B$ là đỉnh cột đèn. Kẻ $MH$ vuông góc với $AB$ tại $H$. Khi đó $MH$ song song với mặt đất, $MH=12$ m và $AH=1{,}5$ m.

Tam giác $MHB$ vuông tại $H$, $\widehat{HMB}=30^\circ$, cạnh đối của góc này là $HB$ và cạnh kề là $MH$. Do đó
\[
\tan 30^\circ=\dfrac{HB}{MH}
\Rightarrow HB=MH\cdot\tan 30^\circ=12\cdot\dfrac{\sqrt{3}}{3}=4\sqrt{3}\ \text{(m)}.
\]
Chiều cao cột đèn là
\[
AB=AH+HB=1{,}5+4\sqrt{3}\ \text{(m)}.
\]
Vì $\sqrt{3}\approx 1{,}732$ nên $4\sqrt{3}\approx 6{,}928$ và
\[
AB\approx 1{,}5+6{,}928=8{,}428\approx 8{,}4\ \text{(m)}.
\]
Vậy chiều cao cột đèn khoảng $8{,}4$ m.

\item Phần cột đèn nằm phía trên tầm mắt vẫn là $HB=4\sqrt{3}$ m.

Gọi $d$ (m) là khoảng cách từ học sinh đến gốc cột đèn, $d>0$. Tia nhìn tạo với phương nằm ngang góc $45^\circ$ nên
\[
\tan 45^\circ=\dfrac{4\sqrt{3}}{d}.
\]
Vì $\tan 45^\circ=1$ nên $d=4\sqrt{3}$.

Vậy học sinh phải đứng cách gốc cột đèn $4\sqrt{3}$ m.
\end{enumerate}
}
\end{bt}

\begin{enumerate}
    \item Chứng minh rằng bốn điểm $A, P, M, O$ cùng thuộc một đường tròn và $BM \parallel OP$.
    \item Đường thẳng vuông góc với $AB$ tại $O$ cắt tia $BM$ tại $N$. Chứng minh tứ giác $OBNP$ là hình bình hành.
