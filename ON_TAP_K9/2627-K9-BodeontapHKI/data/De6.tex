\begin{bt}[1,5 điểm]
Giải các phương trình, bất phương trình và hệ phương trình sau:
\begin{listEX}[3]
\item $\dfrac{3x+5}{x-2}-\dfrac{3x-1}{x+2}=\dfrac{26}{x^2-4}$
\item $\dfrac{2x-3}{3}-\dfrac{5x+1}{4}\ge\dfrac{1-2x}{6}$
\item $\begin{cases}
2(x-1)-3(y+2)=-11\\
\dfrac{x+2y}{3}+\dfrac{x-y}{2}=3
\end{cases}$
\end{listEX}

\loigiai{
\begin{enumerate}
\item Điều kiện xác định: $x\ne 2$, $x\ne -2$.
\begin{align*}
\dfrac{3x+5}{x-2}-\dfrac{3x-1}{x+2}&=\dfrac{26}{x^2-4}\\
(3x+5)(x+2)-(3x-1)(x-2)&=26\\
18x+8&=26\\
x&=1.
\end{align*}
Giá trị $x=1$ thỏa mãn điều kiện xác định. Vậy phương trình có nghiệm $x=1$.

\item Nhân cả hai vế với $12>0$, ta được
\begin{align*}
4(2x-3)-3(5x+1)&\ge 2(1-2x)\\
8x-12-15x-3&\ge 2-4x\\
-3x&\ge 17\\
x&\le -\dfrac{17}{3}.
\end{align*}
Vậy bất phương trình có nghiệm $x\le -\dfrac{17}{3}$.

\item Dùng phương pháp thế, ta có
\begin{align*}
&\begin{cases}
2(x-1)-3(y+2)=-11,\\
\dfrac{x+2y}{3}+\dfrac{x-y}{2}=3
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
2x-3y=-3,\\
5x+y=18
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
2x-3(18-5x)=-3,\\
y=18-5x
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
17x=51,\\
y=18-5x
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
x=3,\\
y=3.
\end{cases}
\end{align*}
Vậy hệ phương trình có nghiệm $(x;y)=(3;3)$.
\end{enumerate}
}
\end{bt}

\begin{bt}[1 điểm]
Rút gọn các biểu thức sau:
\begin{listEX}[2]
\item $\sqrt{14+6\sqrt{5}}-\sqrt{17-12\sqrt{2}}$
\item $\dfrac{\sqrt{45}+\sqrt{5}}{\sqrt{5}-\sqrt{3}}
-\dfrac{4\sqrt{3}}{\sqrt{5}+\sqrt{3}}$
\end{listEX}
\loigiai{
\begin{enumerate}
\setlength{\itemsep}{1em}

\item Ta có
\begin{align*}
14+6\sqrt{5}
&=9+2\cdot3\sqrt{5}+5\\
&=\left(3+\sqrt{5}\right)^2,\\
17-12\sqrt{2}
&=9-2\cdot3\cdot2\sqrt{2}+8\\
&=\left(3-2\sqrt{2}\right)^2.
\end{align*}
Vì $3-2\sqrt{2}>0$ nên
\begin{align*}
\sqrt{14+6\sqrt{5}}-\sqrt{17-12\sqrt{2}}
&=\sqrt{\left(3+\sqrt{5}\right)^2}
-\sqrt{\left(3-2\sqrt{2}\right)^2}\\
&=3+\sqrt{5}-\left(3-2\sqrt{2}\right)\\
&=\sqrt{5}+2\sqrt{2}.
\end{align*}
Vậy biểu thức đã cho rút gọn thành $\sqrt{5}+2\sqrt{2}$.

\item Ta có $\sqrt{45}=3\sqrt{5}$, do đó
\begin{align*}
\dfrac{\sqrt{45}+\sqrt{5}}{\sqrt{5}-\sqrt{3}}
-\dfrac{4\sqrt{3}}{\sqrt{5}+\sqrt{3}}\\
&=\dfrac{4\sqrt{5}}{\sqrt{5}-\sqrt{3}}
-\dfrac{4\sqrt{3}}{\sqrt{5}+\sqrt{3}}\\
&=\dfrac{4\sqrt{5}\left(\sqrt{5}+\sqrt{3}\right)}{5-3}
-\dfrac{4\sqrt{3}\left(\sqrt{5}-\sqrt{3}\right)}{5-3}\\
&=10+2\sqrt{15}-\left(2\sqrt{15}-6\right)\\
&=16.
\end{align*}
Vậy biểu thức đã cho rút gọn thành $16$.
\end{enumerate}
}
\end{bt}

\begin{bt}[1 điểm]
Một công ty du lịch thông báo giá thuê xe cho chuyến đi một chiều như sau:
\begin{itemize}
\item Hãng A: giá mở cửa $200$ nghìn đồng và thêm $4$ nghìn đồng cho mỗi kilômét.
\item Hãng B: giá mở cửa $80$ nghìn đồng và thêm $6$ nghìn đồng cho mỗi kilômét.
\end{itemize}
Gọi $x$ (km), với $x>0$, là quãng đường chuyến đi. Gọi $f(x)$ và $g(x)$ (nghìn đồng) lần lượt là số tiền phải trả cho hãng A và hãng B.
\begin{enumerate}
\item Viết công thức của $f(x)$ và $g(x)$.
\item Tìm quãng đường để số tiền thuê hai hãng bằng nhau. Nếu quãng đường dài hơn mức vừa tìm thì nên chọn hãng nào để tiết kiệm chi phí? Vì sao?
\end{enumerate}
\loigiai{
\begin{enumerate}
\item Vì hãng A thu $200$ nghìn đồng tiền mở cửa và $4$ nghìn đồng cho mỗi kilômét nên
\[
f(x)=4x+200.
\]
Vì hãng B thu $80$ nghìn đồng tiền mở cửa và $6$ nghìn đồng cho mỗi kilômét nên
\[
g(x)=6x+80.
\]

\item Hai hãng có số tiền thuê bằng nhau khi $f(x)=g(x)$. Ta có
\begin{align*}
4x+200&=6x+80\\
200-80&=6x-4x\\
120&=2x\\
x&=60.
\end{align*}
Vậy hai hãng thu cùng số tiền khi quãng đường dài $60$ km.

Xét hiệu
\begin{align*}
f(x)-g(x)
&=(4x+200)-(6x+80)\\
&=120-2x\\
&=2(60-x).
\end{align*}
Với $x>60$ thì $60-x<0$, nên $f(x)-g(x)<0$, tức là $f(x)<g(x)$.

Do đó, nếu quãng đường dài hơn $60$ km thì nên chọn hãng A để tiết kiệm chi phí.
\end{enumerate}
}
\end{bt}

\begin{bt}[1 điểm]
Mẹ Hoa nhờ bạn mua một nồi cơm điện và một bàn là. Giá niêm yết của hai món cộng lại là $750\,000$ đồng. Hôm nay nồi cơm điện được giảm $20\%$, bàn là được giảm $10\%$ so với giá niêm yết, nên số tiền phải trả ít hơn tổng giá niêm yết $120\,000$ đồng.
\begin{enumerate}
\item Tìm giá niêm yết của mỗi món.
\item Tính số tiền thực tế phải trả cho mỗi món sau khi giảm giá.
\end{enumerate}
\loigiai{
\begin{enumerate}
\item Gọi $x$ (nghìn đồng) là giá niêm yết của nồi cơm điện và $y$ (nghìn đồng) là giá niêm yết của bàn là, với $x>0$, $y>0$.

Tổng giá niêm yết là $750$ nghìn đồng, nên
\[
x+y=750. \tag{1}
\]
Số tiền được giảm là $20\%$ giá nồi cộng với $10\%$ giá bàn là và bằng $120$ nghìn đồng, nên
\[
\dfrac{20}{100}x+\dfrac{10}{100}y=120,
\]
tức là
\[
0{,}2x+0{,}1y=120.
\]
Nhân hai vế với $10$, ta được
\[
2x+y=1200. \tag{2}
\]
Lấy $(2)$ trừ $(1)$:
\begin{align*}
(2x+y)-(x+y)&=1200-750\\
x&=450.
\end{align*}
Thay $x=450$ vào $(1)$:
\[
450+y=750 \Rightarrow y=300.
\]
Cặp $(x;y)=(450;300)$ thỏa điều kiện $x>0$, $y>0$.

Vậy giá niêm yết của nồi cơm điện là $450\,000$ đồng và giá niêm yết của bàn là là $300\,000$ đồng.

\item Sau khi giảm giá, số tiền thực tế là
\begin{align*}
\text{nồi cơm điện: }&450\,000\cdot(100\%-20\%)=450\,000\cdot 0{,}8=360\,000 \text{ (đồng)},\\
\text{bàn là: }&300\,000\cdot(100\%-10\%)=300\,000\cdot 0{,}9=270\,000 \text{ (đồng)}.
\end{align*}
Vậy phải trả $360\,000$ đồng cho nồi cơm điện và $270\,000$ đồng cho bàn là.

\emph{Kiểm tra.} Số tiền giảm là $90\,000+30\,000=120\,000$ (đồng), đúng với giả thiết.
\end{enumerate}
}
\end{bt}

\begin{bt}[3,0 điểm]
Cho nửa đường tròn $(O)$ đường kính $AB$. Từ $A$ và $B$ kẻ hai tiếp tuyến $Ax$, $By$. Qua điểm $M$ thuộc nửa đường tròn, kẻ tiếp tuyến thứ ba cắt các tiếp tuyến $Ax$, $By$ lần lượt tại $C$ và $D$. Các đường thẳng $AD$ và $BC$ cắt nhau tại $N$ (hình vẽ).
\begin{enumerate}
\item Chứng minh rằng $\widehat{COD}=90^\circ$ và $AC\cdot BD=\dfrac{AB^2}{4}$.
\item Chứng minh rằng $MN\perp AB$.
\item Cho $OD=2OM$. Tính diện tích hình quạt giới hạn bởi các bán kính $OM$, $OA$ và cung nhỏ $MA$.
\end{enumerate}

\loigiai{
\begin{center}
\begin{tikzpicture}[scale=1.5, font=\footnotesize, line join=round, line cap=round]
	\def\r{2}
	\def\ang{75}
	\pgfmathsetmacro{\cy}{\r*cot(\ang/2)}
	\pgfmathsetmacro{\dy}{\r*tan(\ang/2)}
	\pgfmathsetmacro{\mx}{\r*cos(\ang)}
	\pgfmathsetmacro{\my}{\r*sin(\ang)}
	\coordinate (O) at (0,0);
	\coordinate (A) at (-\r,0);
	\coordinate (B) at (\r,0);
	\coordinate (M) at (\mx,\my);
	\coordinate (C) at (-\r,\cy);
	\coordinate (D) at (\r,\dy);
	\coordinate (N) at (\mx,\my/2);
	\coordinate (H) at (\mx,0);
	\coordinate (Px) at (-\r,\cy+0.42);
	\coordinate (Py) at (\r,\cy+0.42);
	\fill[mauChuDao!8] (A) arc (180:0:\r) -- cycle;
	\draw[thick] (A) arc (180:0:\r);
	\draw[thick] (A)--(B);
	\draw[thick] (A)--(Px) (B)--(Py);
	\draw[thick] (C)--(D);
	\draw (A)--(D) (B)--(C);
	\draw (O)--(M);
	\draw[dashed] (O)--(C) (O)--(D) (M)--(H);
	\draw ($(H)+(-0.16,0)$)--++(0,0.16)--++(0.16,0);
	\foreach \p/\pos in {A/below left, B/below right, C/left, D/right}
		\fill (\p) circle (1.15pt) node[\pos] {$\p$};
	\fill (O) circle (1.15pt) node[below left=1pt] {$O$};
	\fill (H) circle (1.15pt) node[below right=1pt] {$H$};
	\fill (M) circle (1.15pt) node[above=2pt] {$M$};
	\fill (N) circle (1.15pt) node[right=5pt, inner sep=1.2pt] {$N$};
	\node[above left] at (Px) {$x$};
	\node[above right] at (Py) {$y$};
\end{tikzpicture}
\end{center}
\begin{enumerate}
\setlength{\itemsep}{0.85em}
\item Gọi $R$ là bán kính nửa đường tròn. Khi đó $OA=OB=OM=R$ và $AB=2R$.

Vì $Ax$ là tiếp tuyến tại $A$ nên $OA\perp Ax$. Vì $By$ là tiếp tuyến tại $B$ nên $OB\perp By$. Vì tiếp tuyến tại $M$ tiếp xúc với nửa đường tròn tại $M$ nên $OM\perp CD$.

Hai tiếp tuyến xuất phát từ cùng một điểm thì bằng nhau, nên $CA=CM$ và $DB=DM$.

Xét $\triangle OAC$ và $\triangle OMC$:
\[
OA=OM,\qquad CA=CM,\qquad OC\text{ chung}.
\]
Suy ra $\triangle OAC\backsim\triangle OMC$ (c.c.c), nên $\widehat{AOC}=\widehat{COM}$.

Hai góc này kề nhau và $\widehat{AOC}+\widehat{COM}=\widehat{AOM}$, nên $OC$ là tia phân giác của $\widehat{AOM}$.

Tương tự, $\triangle OBD\backsim\triangle OMD$ (c.c.c), nên $OD$ là tia phân giác của $\widehat{BOM}$.

Vì $A$, $O$, $B$ thẳng hàng nên $\widehat{AOM}+\widehat{BOM}=180^\circ$. Do đó
\[
\widehat{COD}
=\widehat{COM}+\widehat{MOD}
=\dfrac12\widehat{AOM}+\dfrac12\widehat{BOM}
=\dfrac12\cdot 180^\circ
=90^\circ.
\]

Đặt $\widehat{AOC}=\widehat{COM}=\alpha$ và $\widehat{BOD}=\widehat{DOM}=\beta$. Khi đó $2\alpha+2\beta=180^\circ$, nên $\alpha+\beta=90^\circ$ và $\beta=90^\circ-\alpha$.

Tam giác $OAC$ vuông tại $A$, nên
\[
\tan\alpha=\dfrac{AC}{OA}=\dfrac{AC}{R}\Rightarrow AC=R\tan\alpha.
\]
Tam giác $OBD$ vuông tại $B$, nên
\[
\tan\beta=\dfrac{BD}{OB}=\dfrac{BD}{R}.
\]
Vì $\tan(90^\circ-\alpha)=\cot\alpha$ nên $BD=R\cot\alpha$.

Do đó
\[
AC\cdot BD=R\tan\alpha\cdot R\cot\alpha=R^2.
\]
Mặt khác
\[
\dfrac{AB^2}{4}=\dfrac{(2R)^2}{4}=R^2.
\]
Vậy $\widehat{COD}=90^\circ$ và $AC\cdot BD=\dfrac{AB^2}{4}$.

\item Kẻ $MH\perp AB$ tại $H$. Ta chứng minh $N$ nằm trên $MH$.

Vì $AB$ là đường kính nên $\widehat{AMB}=90^\circ$. Trong tam giác vuông $AMB$, đường cao $MH$ ứng với cạnh huyền $AB$ nên
\[
AM^2=AH\cdot AB,\qquad BM^2=BH\cdot AB.
\]
Suy ra $\dfrac{AH}{BH}=\dfrac{AM^2}{BM^2}$.

Theo câu a), $\widehat{AOM}=2\alpha$ và $\widehat{BOM}=2\beta$ với $\beta=90^\circ-\alpha$.

Tam giác $OAM$ cân tại $O$. Đường phân giác của $\widehat{AOM}$ cắt $AM$ tại trung điểm $F$ của $AM$ và vuông góc với $AM$. Tam giác $OAF$ vuông tại $F$, có góc ở $O$ bằng $\alpha$ và cạnh huyền $OA=R$, nên
\[
\sin\alpha=\dfrac{AF}{R}=\dfrac{AM}{2R}.
\]
Suy ra $AM=2R\sin\alpha$.

Chứng minh tương tự, $BM=2R\sin\beta$. Vì $\beta=90^\circ-\alpha$ nên $\sin\beta=\cos\alpha$, do đó $BM=2R\cos\alpha$.

Từ đó
\[
\dfrac{AH}{BH}
=\dfrac{AM^2}{BM^2}
=\dfrac{(2R\sin\alpha)^2}{(2R\cos\alpha)^2}
=\tan^2\alpha.
\]

Theo câu a), $AC=R\tan\alpha$ và $BD=R\cot\alpha$, nên
\[
\dfrac{AC}{BD}
=\dfrac{R\tan\alpha}{R\cot\alpha}
=\tan^2\alpha.
\]
Vậy
\[
\dfrac{AH}{BH}=\dfrac{AC}{BD}. \tag{1}
\]

Vì $OA\perp Ax$, $OB\perp By$ và $A$, $O$, $B$ thẳng hàng nên $Ax\parallel By$. Do đó $AC\parallel BD$.

Xét $\triangle ACN$ và $\triangle DBN$:
\begin{itemize}
\item $\widehat{CAN}=\widehat{BDN}$ (so le trong, $AC\parallel BD$, cát tuyến $AD$);
\item $\widehat{ACN}=\widehat{DBN}$ (so le trong, $AC\parallel BD$, cát tuyến $BC$).
\end{itemize}
Suy ra $\triangle ACN\backsim\triangle DBN$ (g.g). Do đó
\[
\dfrac{AN}{DN}=\dfrac{AC}{BD}.
\]
Kết hợp với $(1)$, ta được
\[
\dfrac{AN}{DN}=\dfrac{AH}{BH}.
\]
Suy ra
\begin{align*}
\dfrac{AN}{AD}
&=\dfrac{AN}{AN+DN}
=\dfrac{1}{1+\dfrac{DN}{AN}}
=\dfrac{1}{1+\dfrac{BH}{AH}}\\
&=\dfrac{AH}{AH+BH}
=\dfrac{AH}{AB}. \tag{2}
\end{align*}

Gọi $\ell$ là đường thẳng đi qua $N$ và vuông góc với $AB$, cắt $AB$ tại $P$. Ba đường thẳng $Ax$, $\ell$ và $By$ cùng vuông góc với $AB$ nên song song với nhau. Cát tuyến $AD$ cắt ba đường thẳng này lần lượt tại $A$, $N$, $D$; cát tuyến $AB$ cắt chúng lần lượt tại $A$, $P$, $B$.

Theo định lý Thalès,
\[
\dfrac{AN}{AD}=\dfrac{AP}{AB}.
\]
Kết hợp với $(2)$ suy ra $AP=AH$. Vì $P$ và $H$ cùng nằm trên đoạn thẳng $AB$ nên $P$ trùng $H$.

Vậy $\ell$ chính là đường thẳng $MH$, nên $N$ nằm trên $MH$. Do đó $MN\perp AB$.

\item Theo câu a), tam giác $OBD$ vuông tại $B$ và $OD$ là phân giác của $\widehat{BOM}$.

Vì $OD=2OM=2R$ và $OB=R$ nên
\[
\cos\widehat{BOD}=\dfrac{OB}{OD}=\dfrac{R}{2R}=\dfrac12.
\]
Góc $\widehat{BOD}$ là góc nhọn, nên $\widehat{BOD}=60^\circ$.

Suy ra
\[
\widehat{BOM}=2\widehat{BOD}=120^\circ.
\]
Vì $A$, $O$, $B$ thẳng hàng nên
\[
\widehat{AOM}=180^\circ-\widehat{BOM}=60^\circ.
\]

Do $60^\circ<180^\circ$, cung nhỏ $MA$ là cung chắn góc ở tâm $\widehat{AOM}=60^\circ$. Hình quạt giới hạn bởi hai bán kính $OA$, $OM$ và cung nhỏ $MA$ có bán kính $R$ và góc ở tâm $60^\circ$. Diện tích hình quạt là
\begin{align*}
S
&=\dfrac{\pi R^2\cdot 60}{360}
=\dfrac{\pi R^2}{6}
=\dfrac{\pi\cdot OM^2}{6}.
\end{align*}
Vậy diện tích hình quạt cần tính là $\dfrac{\pi\cdot OM^2}{6}$.
\end{enumerate}
}
