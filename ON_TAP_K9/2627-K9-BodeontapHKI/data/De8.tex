\begin{bt}[1,5 điểm]
Giải các phương trình, bất phương trình và hệ phương trình sau:
\begin{listEX}[3]
\item $\dfrac{x+2}{x-1}+\dfrac{x-3}{x+1}=\dfrac{4x+2}{x^2-1}$
\item $\dfrac{2x-3}{3}-\dfrac{x+4}{2}\le\dfrac{5(x-2)}{6}$
\item $\begin{cases}
(x+2)(y-1)=xy-6\\
(x-1)(y+3)=xy+4
\end{cases}$
\end{listEX}
\loigiai{
\begin{enumerate}
\item Điều kiện xác định: $x\ne 1$, $x\ne -1$.
\begin{align*}
\dfrac{x+2}{x-1}+\dfrac{x-3}{x+1}&=\dfrac{4x+2}{x^2-1}\\
(x+2)(x+1)+(x-3)(x-1)&=4x+2\\
2x^2-5x+3&=0\\
(2x-3)(x-1)&=0.
\end{align*}
Suy ra $x=\dfrac{3}{2}$ hoặc $x=1$. \\
Vì $x=1$ không thỏa mãn điều kiện xác định nên phương trình có nghiệm $x=\dfrac{3}{2}$.

\item Nhân cả hai vế với $6>0$, ta được
\begin{align*}
2(2x-3)-3(x+4)&\le 5(x-2)\\
x-18&\le 5x-10\\
-8&\le 4x\\
x&\ge -2.
\end{align*}
Vậy bất phương trình có nghiệm $x\ge -2$.

\item Dùng phương pháp thế, ta có
\begin{align*}
&\begin{cases}
(x+2)(y-1)=xy-6,\\
(x-1)(y+3)=xy+4
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
x-2y=4,\\
3x-y=7
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
x=4+2y,\\
3x-y=7
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
x=4+2y,\\
3(4+2y)-y=7
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
x=4+2y,\\
5y=-5
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
x=2,\\
y=-1.
\end{cases}
\end{align*}
Vậy hệ phương trình có nghiệm $(x;y)=(2;-1)$.
\end{enumerate}
}
\end{bt}

\begin{bt}[1 điểm]
Rút gọn các biểu thức sau:
\begin{listEX}[1]
\item $A=\sqrt{x+6\sqrt{x}+9}+\sqrt{x-4\sqrt{x}+4}
-\sqrt{x-10\sqrt{x}+25}$, với $4\le x\le 25$.

\item $B=\left(\dfrac{\sqrt{x}+2}{\sqrt{x}-2}
-\dfrac{\sqrt{x}-2}{\sqrt{x}+2}\right)
\cdot\dfrac{x-4}{8}$, với $x\ge 0,\ x\ne 4$.
\end{listEX}
\loigiai{
\begin{enumerate}
\setlength{\itemsep}{1em}

\item Ta có
\begin{align*}
x+6\sqrt{x}+9&=\left(\sqrt{x}+3\right)^2,\\
x-4\sqrt{x}+4&=\left(\sqrt{x}-2\right)^2,\\
x-10\sqrt{x}+25&=\left(\sqrt{x}-5\right)^2.
\end{align*}
Vì $4\le x\le25$ nên $2\le\sqrt{x}\le5$. Do đó
\begin{align*}
A
&=\left|\sqrt{x}+3\right|
+\left|\sqrt{x}-2\right|
-\left|\sqrt{x}-5\right|\\
&=\sqrt{x}+3+\sqrt{x}-2-\left(5-\sqrt{x}\right)\\
&=3\sqrt{x}-4.
\end{align*}
Vậy $A=3\sqrt{x}-4$.

\item Với $x\ge0$ và $x\ne4$, ta có
\begin{align*}
B
&=\left(\dfrac{\sqrt{x}+2}{\sqrt{x}-2}
-\dfrac{\sqrt{x}-2}{\sqrt{x}+2}\right)
\cdot\dfrac{x-4}{8}\\
&=\dfrac{\left(\sqrt{x}+2\right)^2
-\left(\sqrt{x}-2\right)^2}{x-4}
\cdot\dfrac{x-4}{8}\\
&=\dfrac{8\sqrt{x}}{x-4}\cdot\dfrac{x-4}{8}\\
&=\sqrt{x}.
\end{align*}
Vậy $B=\sqrt{x}$.
\end{enumerate}
}
\end{bt}

\begin{bt}[1,0 điểm]
Một rạp chiếu phim bán vé người lớn và vé trẻ em. Buổi sáng, rạp bán $3$ vé người lớn và $2$ vé trẻ em, thu được $400\,000$ đồng. Buổi chiều, giá vé người lớn giảm $25\%$ và giá vé trẻ em giảm $10\%$ so với buổi sáng; rạp bán $4$ vé người lớn và $2$ vé trẻ em, thu được $390\,000$ đồng. Biết trong mỗi buổi, các vé cùng loại có cùng giá. Tính giá một vé người lớn và một vé trẻ em ở buổi sáng.
\loigiai{
Gọi $x$ (nghìn đồng) là giá một vé người lớn buổi sáng và $y$ (nghìn đồng) là giá một vé trẻ em buổi sáng, với $x>0$, $y>0$.

Buổi sáng cho ta
\[
3x+2y=400. \tag{1}
\]
Buổi chiều, vé người lớn có giá $\dfrac{3}{4}x$ và vé trẻ em có giá $\dfrac{9}{10}y$, nên
\[
4\cdot\dfrac{3}{4}x+2\cdot\dfrac{9}{10}y=390,
\]
tức là
\[
3x+\dfrac{9}{5}y=390.
\]
Nhân hai vế với $5$:
\[
15x+9y=1950.
\]
Chia hai vế cho $3$:
\[
5x+3y=650. \tag{2}
\]
Nhân $(2)$ với $2$ và nhân $(1)$ với $3$:
\begin{align*}
10x+6y&=1300, \tag{3}\\
9x+6y&=1200. \tag{4}
\end{align*}
Lấy $(3)$ trừ $(4)$:
\[
x=100.
\]
Thay vào $(1)$:
\[
3\cdot 100+2y=400 \Rightarrow 2y=100 \Rightarrow y=50.
\]
Cặp $(x;y)=(100;50)$ thỏa $x>0$, $y>0$.

\emph{Kiểm tra.} Buổi chiều, vé người lớn giá $75$ nghìn đồng và vé trẻ em giá $45$ nghìn đồng.
\[
4\cdot 75+2\cdot 45=300+90=390,
\]
đúng với giả thiết.

Vậy buổi sáng, một vé người lớn giá $100\,000$ đồng và một vé trẻ em giá $50\,000$ đồng.
}
\end{bt}

\item Biết diện tích phần tô đậm là $32-4\pi$ (cm$^2$). Tìm $x$.

\begin{tikzpicture}[scale=0.8, font=\scriptsize, line join=round]

\item Diện tích hình chữ nhật là
\[
S_{\text{cn}}=(x+5)(x+1)=x^2+6x+5\ \text{(cm}^2\text{)}.
\]

\[
S_{\text{tròn}}=\pi\cdot 2^2=4\pi\ \text{(cm}^2\text{)}.
\]
Diện tích phần tô đậm là
\[
S=S_{\text{cn}}-S_{\text{tròn}}=x^2+6x+5-4\pi\ \text{(cm}^2\text{)}.
\]

\item Ta có
\begin{align*}
x^2+6x+5-4\pi&=32-4\pi\\
x^2+6x+5&=32\\
x^2+6x-27&=0\\
(x+9)(x-3)&=0.
\end{align*}
Suy ra $x=-9$ hoặc $x=3$.

Vậy $x=3$ cm.
\end{enumerate}
}
\end{bt}

\begin{bt}[3,0 điểm]
	Cho $(O;R)$ và điểm $A$ nằm ngoài đường tròn. Vẽ các tiếp tuyến $AB$, $AC$ với $(O)$ ($B$, $C$ là hai tiếp điểm).
	\begin{enumerate}
	\item Chứng minh rằng $OA\perp BC$.
	\item Gọi $H$ là giao điểm của $OA$ và $BC$. Chứng minh rằng
	\[
	4OH\cdot HA=BC^2.
	\]
	\item Gọi $M$ và $N$ lần lượt là trung điểm của $AB$, $AC$. Trên cung nhỏ $BC$ lấy điểm $D$ sao cho tiếp tuyến tại $D$ của $(O)$ cắt tia $MN$ tại $I$. Gọi $T$ là hình chiếu của $D$ trên $OI$. Chứng minh rằng
	\[
	OH\cdot OA=OT\cdot OI
	\quad \text{và} \quad
	IA=ID.
	\]
	\end{enumerate}
	
	\loigiai{
	\begin{center}
	\begin{tikzpicture}[scale=1.1, font=\footnotesize, line join=round, line cap=round]
		% Các điểm
		\coordinate (O) at (0,0);
		\coordinate (A) at (0,4);
		\coordinate (B) at (-1.732,1);
		\coordinate (C) at (1.732,1);
		\coordinate (H) at (0,1);
		\coordinate (M) at (-0.866,2.5);
		\coordinate (N) at (0.866,2.5);
		\coordinate (D) at (-0.684,1.879);
		\coordinate (I) at (1.020,2.5);
		\coordinate (T) at (0.560,1.372);
	
		% Đường tròn
		\draw[thick] (O) circle (2);
	
		% Các đoạn chính
		\draw[thick] (A)--(B) (A)--(C);
		\draw[thick] (B)--(C);
		\draw[thick] (O)--(A);
		\draw[thick] (M)--(I);
		\draw[thick] (D)--(I);
	
		% Các đoạn phụ
		\draw[dashed] (O)--(B) (O)--(C) (O)--(D) (O)--(I) (D)--(T);
	
		% Dấu vuông góc
		\draw ($(T)+(0.08,0.03)$)--++(-0.09,0.12)--++(-0.12,-0.09);
		\draw ($(H)+(-0.08,0)$)--++(0,0.08)--++(0.08,0);
	
		% Điểm và nhãn
		\fill (A) circle (1.2pt) node[above] {$A$};
		\fill (O) circle (1.2pt) node[left] {$O$};
		\fill (B) circle (1.2pt) node[left] {$B$};
		\fill (C) circle (1.2pt) node[right] {$C$};
		\fill (H) circle (1.2pt) node[right] {$H$};
		\fill (M) circle (1.2pt) node[left] {$M$};
		\fill (N) circle (1.2pt) node[above] {$N$};
		\fill (D) circle (1.2pt) node[above left] {$D$};
		\fill (I) circle (1.2pt) node[right] {$I$};
		\fill (T) circle (1.2pt) node[right] {$T$};
	\end{tikzpicture}
	\end{center}
	
	\begin{enumerate}
	\item Vì $AB$, $AC$ là hai tiếp tuyến kẻ từ $A$ đến đường tròn $(O)$ nên
	\[
	AB=AC.
	\]
	Mặt khác,
	\[
	OB=OC
	\]
	(vì cùng bằng bán kính của đường tròn $(O)$).
	
	Vậy $A$ và $O$ đều cách đều hai điểm $B$, $C$, suy ra đường thẳng $AO$ là đường trung trực của đoạn thẳng $BC$.
	
	Do đó
	\[
	OA\perp BC.
	\]
	
	\item Từ câu a), ta có $OA\perp BC$. Lại do $AO$ là đường trung trực của $BC$ nên
	\[
	HB=HC=\dfrac{BC}{2}. \tag{1}
	\]
	
	Xét hai tam giác vuông $\triangle AHB$ và $\triangle AOB$, ta có:
	\begin{itemize}
	\item $\widehat{AHB}=90^\circ$;
	\item $\widehat{ABO}=90^\circ$ (bán kính vuông góc với tiếp tuyến);
	\item $\widehat{BAH}=\widehat{BAO}$.
	\end{itemize}
	Suy ra
	\[
	\triangle AHB \sim \triangle AOB.
	\]
	
	Do đó
	\[
	\dfrac{AH}{AB}=\dfrac{AB}{AO}
	\Rightarrow AH\cdot AO=AB^2. \tag{2}
	\]
	
	Mà trong tam giác vuông $AOB$, ta có
	\[
	AB^2=AO^2-OB^2=AO^2-R^2.
	\]
	Thay vào (2), được
	\[
	AH\cdot AO=AO^2-R^2.
	\]
	Suy ra
	\[
	AO(AO-AH)=R^2.
	\]
	Vì $O,H,A$ thẳng hàng và $H$ nằm giữa $O,A$ nên
	\[
	AO-AH=OH.
	\]
	Do đó
	\[
	OA\cdot OH=R^2. \tag{3}
	\]
	
	Xét tam giác vuông $OHB$, ta có
	\[
	HB^2=OB^2-OH^2=R^2-OH^2.
	\]
	Theo (3), $R^2=OA\cdot OH$, nên
	\[
	HB^2=OA\cdot OH-OH^2=OH(OA-OH).
	\]
	Mà $OA-OH=HA$, suy ra
	\[
	HB^2=OH\cdot HA.
	\]
	Nhân hai vế với $4$, ta được
	\[
	4HB^2=4OH\cdot HA.
	\]
	Theo (1),
	\[
	BC=2HB \Rightarrow BC^2=4HB^2.
	\]
	Vậy
	\[
	4OH\cdot HA=BC^2.
	\]
	
	\item Gọi $K=OA\cap MN$.
	
	Vì $M$, $N$ lần lượt là trung điểm của $AB$, $AC$ nên trong tam giác $ABC$ ta có
	\[
	MN\parallel BC.
	\]
	Suy ra
	\[
	OA\perp MN \tag{4}
	\]
	(do $OA\perp BC$ theo câu a)).
	
	Lại vì $MN\parallel BC$, nên $\triangle AMN \sim \triangle ABC$ với tỉ số đồng dạng
	\[
	\dfrac{AM}{AB}=\dfrac12.
	\]
	Do đó
	\[
	\dfrac{AK}{AH}=\dfrac12
	\Rightarrow AK=\dfrac{AH}{2}.
	\]
	Suy ra $K$ là trung điểm của $AH$, nên
	\[
	AK=KH=\dfrac{AH}{2}. \tag{5}
	\]
	
	\medskip
	\textbf{Chứng minh $OH\cdot OA=OT\cdot OI$.}
	
	Vì $ID$ là tiếp tuyến của $(O)$ tại $D$ nên
	\[
	OD\perp ID.
	\]
	Suy ra tam giác $ODI$ vuông tại $D$.
	
	Lại có $T$ là hình chiếu của $D$ trên $OI$, nên
	\[
	DT\perp OI.
	\]
	Vậy $DT$ là đường cao ứng với cạnh huyền $OI$ của tam giác vuông $ODI$.
	
	Theo hệ thức lượng trong tam giác vuông, ta có
	\[
	OD^2=OT\cdot OI.
	\]
	Mà $OD=R$, nên
	\[
	R^2=OT\cdot OI. \tag{6}
	\]
	
	Mặt khác, theo (3),
	\[
	OH\cdot OA=R^2. \tag{7}
	\]
	
	Từ (6) và (7), suy ra
	\[
	OH\cdot OA=OT\cdot OI.
	\]
	
	\medskip
	\textbf{Chứng minh $IA=ID$.}
	
	Do $I\in MN$ và $K\in MN$, từ (4) suy ra hai tam giác $AKI$ và $OKI$ đều vuông tại $K$.
	
	Áp dụng định lý Py-ta-go, ta có
	\[
	IA^2=IK^2+AK^2, \tag{8}
	\]
	\[
	OI^2=IK^2+OK^2. \tag{9}
	\]
	
	Lấy (9) trừ (8), được
	\[
	OI^2-IA^2=OK^2-AK^2=(OK-AK)(OK+AK). \tag{10}
	\]
	
	Từ (5), vì $K$ là trung điểm của $AH$ và các điểm $O,H,K,A$ thẳng hàng theo thứ tự đó, nên
	\[
	OK-AK=OH,
	\qquad
	OK+AK=OA.
	\]
	Thay vào (10), ta được
	\[
	OI^2-IA^2=OH\cdot OA=R^2. \tag{11}
	\]
	
	Mặt khác, trong tam giác vuông $ODI$, ta có
	\[
	ID^2=OI^2-OD^2=OI^2-R^2. \tag{12}
	\]
	
	Từ (11), suy ra
	\[
	IA^2=OI^2-R^2.
	\]
	Kết hợp với (12), ta được
	\[
	IA^2=ID^2.
	\]
	Suy ra
	\[
	IA=ID.
	\]
	
	Vậy ta đã chứng minh được
	\[
	OH\cdot OA=OT\cdot OI
	\quad \text{và} \quad
	IA=ID.
	\]
	\end{enumerate}
	}
