\begin{bt}[1,5 điểm]
Giải các phương trình, bất phương trình và hệ phương trình sau:
\begin{listEX}[3]
\item $\dfrac{3x+4}{x-1}-\dfrac{x-2}{x+2}=\dfrac{19x+14}{x^2+x-2}$
\item $-2(3x-5)+\dfrac{5x+1}{2}>\dfrac{3-x}{4}$
\item $\begin{cases}
\dfrac{x+2y}{3}-\dfrac{2x-y}{4}=2\\
3(x-y)+2(2x+y)=-9
\end{cases}$
\end{listEX}
\loigiai{
\begin{enumerate}
\item Điều kiện xác định: $x\ne 1$, $x\ne -2$.
\begin{align*}
\dfrac{3x+4}{x-1}-\dfrac{x-2}{x+2}&=\dfrac{19x+14}{x^2+x-2}\\
(3x+4)(x+2)-(x-2)(x-1)&=19x+14\\
2x^2-6x-8&=0\\
(x-4)(x+1)&=0.
\end{align*}
Suy ra $x=4$ hoặc $x=-1$; cả hai giá trị đều thỏa mãn điều kiện xác định. Vậy $S=\{-1;4\}$.

\item Nhân cả hai vế với $4>0$, ta được
\begin{align*}
-8(3x-5)+2(5x+1)&>3-x\\
-14x+42&>3-x\\
39&>13x\\
x&<3.
\end{align*}
Vậy bất phương trình có nghiệm $x<3$.

\item Dùng phương pháp thế, ta có
\begin{align*}
&\begin{cases}
\dfrac{x+2y}{3}-\dfrac{2x-y}{4}=2,\\
3(x-y)+2(2x+y)=-9
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
-2x+11y=24,\\
7x-y=-9
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
-2x+11y=24,\\
y=7x+9
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
-2x+11(7x+9)=24,\\
y=7x+9
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
75x=-75,\\
y=7x+9
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
x=-1,\\
y=2.
\end{cases}
\end{align*}
Vậy hệ phương trình có nghiệm $(x;y)=(-1;2)$.
\end{enumerate}
}
\end{bt}
