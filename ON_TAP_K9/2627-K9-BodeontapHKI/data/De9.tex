\begin{bt}[1,5 điểm]
Giải các phương trình, bất phương trình và hệ phương trình sau:
\begin{listEX}[3]
\item $\dfrac{3x+4}{x-1}-\dfrac{x-2}{x+2}=\dfrac{19x+14}{x^2+x-2}$
\item $-2(3x-5)+\dfrac{5x+1}{2}>\dfrac{3-x}{4}$
\item $\begin{cases}
\dfrac{x+2y}{3}-\dfrac{2x-y}{4}=2\\
3(x-y)+2(2x+y)=-9
\end{cases}$
\end{listEX}
\loigiai{
\begin{enumerate}
\item Điều kiện xác định: $x\ne 1$, $x\ne -2$.
\begin{align*}
\dfrac{3x+4}{x-1}-\dfrac{x-2}{x+2}&=\dfrac{19x+14}{x^2+x-2}\\
(3x+4)(x+2)-(x-2)(x-1)&=19x+14\\
2x^2-6x-8&=0\\
(x-4)(x+1)&=0.
\end{align*}
Suy ra $x=4$ hoặc $x=-1$; cả hai giá trị đều thỏa mãn điều kiện xác định. Vậy $S=\{-1;4\}$.

\item Nhân cả hai vế với $4>0$, ta được
\begin{align*}
-8(3x-5)+2(5x+1)&>3-x\\
-14x+42&>3-x\\
39&>13x\\
x&<3.
\end{align*}
Vậy bất phương trình có nghiệm $x<3$.

\item Dùng phương pháp thế, ta có
\begin{align*}
&\begin{cases}
\dfrac{x+2y}{3}-\dfrac{2x-y}{4}=2,\\
3(x-y)+2(2x+y)=-9
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
-2x+11y=24,\\
7x-y=-9
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
-2x+11y=24,\\
y=7x+9
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
-2x+11(7x+9)=24,\\
y=7x+9
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
75x=-75,\\
y=7x+9
\end{cases}\\[0.7em]
\Leftrightarrow\quad
&\begin{cases}
x=-1,\\
y=2.
\end{cases}
\end{align*}
Vậy hệ phương trình có nghiệm $(x;y)=(-1;2)$.
\end{enumerate}
}
\end{bt}

\begin{bt}[1 điểm]
Rút gọn các biểu thức sau:
\begin{listEX}[1]
\item $A=\sqrt{x+3+4\sqrt{x-1}}+\sqrt{x+3-4\sqrt{x-1}}-\sqrt{4x-4}$, với $1\le x\le 5$.

\item $B=\dfrac{x-4}{\sqrt{x}-2}-\dfrac{x-1}{\sqrt{x}+1}
+\dfrac{2}{\sqrt{x}-2}-\dfrac{2}{\sqrt{x}+2}$, với $x\ge 0,\ x\ne 4$.
\end{listEX}
\loigiai{
\begin{enumerate}
\setlength{\itemsep}{1em}

\item Ta có
\begin{align*}
x+3+4\sqrt{x-1}&=\left(\sqrt{x-1}+2\right)^2,\\
x+3-4\sqrt{x-1}&=\left(\sqrt{x-1}-2\right)^2,\\
4x-4&=\left(2\sqrt{x-1}\right)^2.
\end{align*}
Vì $1\le x\le5$ nên $0\le\sqrt{x-1}\le2$. Do đó
\begin{align*}
A
&=\left|\sqrt{x-1}+2\right|
+\left|\sqrt{x-1}-2\right|
-\left|2\sqrt{x-1}\right|\\
&=\sqrt{x-1}+2+2-\sqrt{x-1}-2\sqrt{x-1}\\
&=4-2\sqrt{x-1}.
\end{align*}
Vậy $A=4-2\sqrt{x-1}$.

\item Với $x\ge0$ và $x\ne4$, ta có
\begin{align*}
B
&=\dfrac{(\sqrt{x}-2)(\sqrt{x}+2)}{\sqrt{x}-2}
-\dfrac{(\sqrt{x}-1)(\sqrt{x}+1)}{\sqrt{x}+1}
+\dfrac{2}{\sqrt{x}-2}-\dfrac{2}{\sqrt{x}+2}\\
&=\sqrt{x}+2-\left(\sqrt{x}-1\right)
+\dfrac{2(\sqrt{x}+2)-2(\sqrt{x}-2)}{x-4}\\
&=3+\dfrac{8}{x-4}\\
&=\dfrac{3(x-4)+8}{x-4}\\
&=\dfrac{3x-4}{x-4}.
\end{align*}
Vậy $B=\dfrac{3x-4}{x-4}$.
\end{enumerate}
}
\end{bt}

\begin{bt}[1,0 điểm]
Một lớp có $36$ học sinh. Chiều cao trung bình của cả lớp là $160$ cm, chiều cao trung bình của các bạn nam là $168$ cm và chiều cao trung bình của các bạn nữ là $156$ cm. Hỏi lớp có bao nhiêu học sinh nam và bao nhiêu học sinh nữ?
\loigiai{
Gọi $x$ là số học sinh nam và $y$ là số học sinh nữ của lớp, với $x,y$ nguyên dương và $x+y=36$.

Tổng chiều cao của cả lớp là $160\cdot 36=5760$ (cm). Tổng chiều cao của các bạn nam là $168x$ (cm) và của các bạn nữ là $156y$ (cm), nên
\[
168x+156y=5760. \tag{1}
\]
Vì $y=36-x$, thay vào $(1)$:
\begin{align*}
168x+156(36-x)&=5760\\
168x+5616-156x&=5760\\
12x&=144\\
x&=12.
\end{align*}
Suy ra $y=36-12=24$.

\emph{Kiểm tra.} Tổng chiều cao là
\[
168\cdot 12+156\cdot 24=2016+3744=5760\ \text{(cm)},
\]
trung bình $5760:36=160$ (cm), đúng với giả thiết.

Vậy lớp có $12$ học sinh nam và $24$ học sinh nữ.
}
\end{bt}

\begin{bt}[1,0 điểm]
\immini[thm]
{Một biển báo gồm một hình chữ nhật và một nửa hình tròn gắn phía trên. Chiều rộng hình chữ nhật bằng đường kính nửa hình tròn và bằng $x$ cm ($x>0$); chiều cao hình chữ nhật là $(x+2)$ cm (hình vẽ).
\begin{enumerate}
\item Viết biểu thức diện tích hình chữ nhật, diện tích nửa hình tròn và diện tích toàn biển báo theo $x$.
\item Biết diện tích toàn biển báo là $48+\dfrac{9\pi}{2}$ (cm$^2$). Tìm $x$.
\end{enumerate}
}
{
\begin{tikzpicture}[scale=0.55, font=\scriptsize, line join=round]
	\fill[gray!25] (0,0) rectangle (3,4);
	\fill[gray!25] (0,4) arc (180:0:1.5) -- cycle;
	\draw (0,0)--(3,0)--(3,4) arc (0:180:1.5) -- cycle;
	\draw[dashed] (0,4)--(3,4);
	\node[below] at (1.5,0) {$x$ cm};
	\node[left] at (0,2) {$(x+2)$ cm};
\end{tikzpicture}
}
\loigiai{
\begin{enumerate}
\item Diện tích hình chữ nhật là
\[
S_{\text{cn}}=x(x+2)=x^2+2x\ \text{(cm}^2\text{)}.
\]
Nửa hình tròn có đường kính $x$ cm nên bán kính $\dfrac{x}{2}$ cm. Diện tích nửa hình tròn là
\[
S_{\text{nửa}}=\dfrac{1}{2}\pi\left(\dfrac{x}{2}\right)^2=\dfrac{\pi x^2}{8}\ \text{(cm}^2\text{)}.
\]
Diện tích toàn biển báo là
\[
S=S_{\text{cn}}+S_{\text{nửa}}=x^2+2x+\dfrac{\pi x^2}{8}\ \text{(cm}^2\text{)}.
\]

\item Ta có
\[
x^2+2x+\dfrac{\pi x^2}{8}=48+\dfrac{9\pi}{2},
\]
nên
\[
\left(x^2+2x-48\right)+\pi\left(\dfrac{x^2}{8}-\dfrac{9}{2}\right)=0.
\]
Phần không chứa $\pi$ và hệ số của $\pi$ đều bằng $0$:
\begin{align*}
\dfrac{x^2}{8}&=\dfrac{9}{2}\\
x^2&=36.
\end{align*}
Vì $x>0$ nên $x=6$.

\emph{Kiểm tra.} $x^2+2x=36+12=48$, đúng với phần không chứa $\pi$.

Vậy $x=6$ cm.
\end{enumerate}
}
\end{bt}

\begin{bt}[1,0 điểm]
Bạn Minh đứng cách một tòa nhà. Mắt bạn cách mặt đất $1{,}5$ m. Trên nóc nhà có một cột ăng-ten thẳng đứng cao $6$ m. Tia nhìn từ mắt đến chân cột tạo với phương nằm ngang một góc $30^\circ$, tia nhìn từ mắt đến đỉnh cột tạo với phương nằm ngang một góc $45^\circ$ (hình vẽ).
\begin{center}
\begin{tikzpicture}[x=0.4cm, y=0.4cm, font=\small, line cap=round, line join=round, >=stealth]
	\pgfmathsetmacro{\d}{9+3*sqrt(3)}
	\pgfmathsetmacro{\yC}{1.5+\d*tan(30)}
	\pgfmathsetmacro{\yB}{1.5+\d}
	\coordinate (G) at (-2.2,0);
	\coordinate (M) at (0,1.5);
	\coordinate (P) at (0,0);
	\coordinate (A) at (\d,0);
	\coordinate (H) at (\d,1.5);
	\coordinate (C) at (\d,\yC);
	\coordinate (B) at (\d,\yB);
	\draw[thick] (G)--(A)--(B);
	\draw[thick] (P)--(M);
	\draw[dashed] (M)--(H);
	\draw (M)--(C);
	\draw (M)--(B);
	\draw ($(H)+(-1.05,0)$)--++(0,1.05)--++(1.05,0);
	\pic[draw, angle radius=1.15cm, angle eccentricity=1.15] {angle = H--M--C};
	\pic[draw, angle radius=1.85cm, angle eccentricity=1.12] {angle = H--M--B};
	\node at ($(M)+(15:3.5)$) {$30^\circ$};
	\node at ($(M)+(24:5.6)$) {$45^\circ$};
	\draw[<->] (-1.55,0)--(-1.55,1.5) node[midway,left] {$1{,}5$ m};
	\draw (-1.55,0)--(P) (-1.55,1.5)--(M);
	\draw[<->] ($(C)+(1.7,0)$)--($(B)+(1.7,0)$) node[midway,right] {$6$ m};
	\draw ($(C)+(1.7,0)$)--(C) ($(B)+(1.7,0)$)--(B);
	\foreach \p/\pos in {M/left,H/right,C/right,B/above right,A/below right}
		\fill (\p) circle (1.15pt) node[\pos] {$\p$};
\end{tikzpicture}
\end{center}
\begin{enumerate}
\item Tính khoảng cách từ Minh đến mặt tường tòa nhà.
\item Tính chiều cao của nóc nhà so với mặt đất.
\end{enumerate}
\textit{(Kết quả viết dưới dạng số chính xác.)}
\loigiai{
\begin{enumerate}
\item Gọi $M$ là vị trí mắt, $C$ là chân cột ăng-ten, $B$ là đỉnh cột và $A$ là chân tường. Kẻ $MH$ vuông góc với $AB$ tại $H$. Khi đó $MH$ song song với mặt đất, $AH=1{,}5$ m và $BC=6$ m.

Đặt $MH=d$ (m), $d>0$.

Tam giác $MHC$ vuông tại $H$, $\widehat{HMC}=30^\circ$, nên
\[
\tan 30^\circ=\dfrac{HC}{d}
\Rightarrow HC=d\cdot\dfrac{\sqrt{3}}{3}.
\]
Tam giác $MHB$ vuông tại $H$, $\widehat{HMB}=45^\circ$, nên
\[
\tan 45^\circ=\dfrac{HB}{d}
\Rightarrow HB=d.
\]
Vì $B$, $C$, $H$ thẳng hàng và $C$ nằm giữa $H$ và $B$ nên $HB=HC+CB$:
\begin{align*}
d&=d\cdot\dfrac{\sqrt{3}}{3}+6\\
d\left(1-\dfrac{\sqrt{3}}{3}\right)&=6\\
d\cdot\dfrac{3-\sqrt{3}}{3}&=6\\
d(3-\sqrt{3})&=18\\
d&=\dfrac{18}{3-\sqrt{3}}.
\end{align*}
Nhân liên hợp với $3+\sqrt{3}$:
\[
d=\dfrac{18(3+\sqrt{3})}{(3-\sqrt{3})(3+\sqrt{3})}
=\dfrac{18(3+\sqrt{3})}{6}
=3(3+\sqrt{3})
=9+3\sqrt{3}.
\]
Vậy Minh đứng cách mặt tường tòa nhà $9+3\sqrt{3}$ m.

\item Chiều cao nóc nhà là
\begin{align*}
AC
&=AH+HC\\
&=1{,}5+d\cdot\dfrac{\sqrt{3}}{3}\\
&=1{,}5+(9+3\sqrt{3})\cdot\dfrac{\sqrt{3}}{3}\\
&=1{,}5+\dfrac{9\sqrt{3}+9}{3}\\
&=1{,}5+3\sqrt{3}+3\\
&=4{,}5+3\sqrt{3}\ \text{(m)}.
\end{align*}
Vậy nóc nhà cao $4{,}5+3\sqrt{3}$ m so với mặt đất.
\end{enumerate}
}
\end{bt}
